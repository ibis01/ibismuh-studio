Rows\n// are pruned opportunistically once older than the 10-minute window —\n// see src/lib/tubeforge/auth-throttle.ts for the law.\nmodel AuthAttempt {\n  id  Int      @id @default(autoincrement())\n  key String // throttle key: the caller's IP (login + change-password share it — a wrong current password IS a login attempt)\n  at  DateTime // the failure instant (explicit, never defaulted — the window math depends on it)\n\n  @@index([key, at])\n}\n\n\n// S102 / P0.4 — the SHARED per-IP rate limiter's fixed-window counters.\n// The middleware's in-memory buckets stay (edge runtime is architecture-\n// banned from the DB — they are coarse per-instance shedding); THIS table is\n// the authoritative multi-instance law, enforced at requireUser: one row per\n// (tiered key, 60s window), atomically incremented via a raw upsert\n// RETURNING count. Survives deploys, shared by every instance, immune to the\n// X-Forwarded-For first-value spoof (trusted-proxy law in shared-rate-limit.ts).\nmodel RateLimitCounter {\n  key         String   // \"<tier>:<client ip>\" — tier is r (read) or w (write)\n  windowStart BigInt   // epoch SECONDS of the fixed 60s window (BigInt: int32 epoch dies in 2038)\n  count       Int      @default(0)\n  updatedAt   DateTime @updatedAt\n\n  @@id([key, windowStart])\n  @@index([updatedAt]) // the sweep scan\n}\n"
