     1→// G1.3 — the graphics composer: static skeleton + ONE applier (S120 §5 /
     2→// Part B). The wb-page.ts seam law applied to the graphics grammar:
     3→//
     4→//   the renderer rasterizes frameAt() through THESE artifacts; the editor
     5→//   plays back frameAt() through THE SAME artifacts. One compose, one
     6→//   applier, two callers — there is no second implementation to drift.
     7→//
     8→// The three artifacts, mirroring whiteboard/wb-page.ts exactly:
     9→//   composeGraphicsScene(scene, doc) → the static HTML/SVG skeleton: the
    10→//     background, one <g id="ge-{id}"> per element in doc order, every
    11→//     piece of per-frame-reactive geometry baked into data-* attributes,
    12→//     static attributes carrying the PLACED state (the poster frame — the
    13→//     applier corrects before any capture).
    14→//   graphicsFramePayload(frame) → the frame truth, serialized for the
    15→//     page (frameAt's GraphicsFrameState, verbatim per element).
    16→//   APPLY_GRAPHICS_FRAME_FN → the ONE applier, a set-only DOM function
    17→//     evaluated by renderer (page.evaluate) and editor (iframe eval)
    18→//     alike. NO CSS animation anywhere — every frame applied by JS
    19→//     (the determinism law, wb-page.ts verbatim).
    20→//
    21→// THE REVEAL CHANNEL — one mechanism per element (the G1.2 contract:
    22→// "wipe/grow/draw expose the reveal-progress channel the composer
    23→// interprets per payload"):
    24→//   bar/barChart → the payload's native mechanism, bar LENGTHS, always
    25→//     baked (progress 1 = full bars; any progress-driving motion —
    26→//     wipe enter, grow exit, emphasis re-trace — modulates them).
    27→//   icon → stroke-dash along the path arc length (svg.ts#pathSegments →
    28→//     totalLength — the S64 geometry, reused, never a second parser),
    29→//     always baked.
    30→//   shape → the FIRST progress-driving motion in slot order (enter →
    31→//     exit → emphasis) picks the mechanism: wipe → a baked clip rect
    32→//     reveals left→right; grow → the shape geometry scales from zero;
    33→//     draw → the outline dashes on (fill appears at progress 1 — the
    34→//     fill-after-outline law). No progress-driving motion → none.
    35→//   text kinds / counter / callout → no mechanism (their legal motion
    36→//     sets drive opacity/offset/scale/count only).
    37→//
    38→// RECORDED v1 DECISIONS (receipts to tune; the names are the contract —
    39→// GRAPHICS_COMPOSE_DEFAULTS is the ONE table):
    40→//   - compose-time defaults: sizePx 48, weight 700, ink #111111, icon
    41→//     strokeWidth 4, shape strokeWidth 4, callout size 28, align center,
    42→//     chart box = ½ surface centered on `at` (types.ts' own comments —
    43→//     the validator clamps, the composer defaults).
    44→//   - bar color is the ink (a per-bar palette is a design-doc slice);
    45→//     bar labels carry the payload unit after the label.
    46→//   - text renders explicit \n lines via tspans; auto-wrap is a design
    47→//     slice, not a drive-by. dominant-baseline centers the block on the
    48→//     anchor; align maps to text-anchor (left→start, center→middle,
    49→//     right→end).
    50→//   - an ABSENT payload composes as an empty element (the validator
    51→//     warned; parse never refuses — the Part F law).
    52→//   - group children nest INSIDE the group's <g> (the group's frame
    53→//     transform/opacity composes with each child's — SVG multiplies
    54→//     subtree opacity for free). Children ride the group's children-array
    55→//     order; an element claimed by two groups lives in the LAST claim
    56→//     (validator dedupes within one group only — recorded semantics).
    57→//   - no camera: graphics is a peer lane; the W5 camera layer stays the
    58→//     whiteboard's. The doc clock IS the timeline.
    59→//   - the applier is IDEMPOTENT BY CONSTRUCTION: every write is an
    60→//     absolute set on a static node — it creates no elements, reads no
    61→//     layout (the W1 getBBox lesson), touches no wall clock, draws no
    62→//     randomness. Full DOM idempotence rides the G1.10 benchmark's
    63→//     frame-hash pin; the unit pins hold the construction laws.
    64→//
    65→// SECURITY (the hostile-icon pin): icon path data is sanitized with
    66→// svg.ts#sanitizeSvg FIRST (payload markup stripped from the d string
    67→// itself), then HTML-attribute-escaped (the hard gate — an escaped value
    68→// cannot break out of its attribute context, ever). Validation counted
    69→// commands on the raw d; the rendered path is the sanitized remainder —
    70→// the budget was an upper bound, sanitize only removes hostile junk.
    71→//
    72→// PURE — no fs, no DOM execution, no clock, no randomness (the S64 law):
    73→// this module builds STRINGS; the DOM owns execution. Leaf-module law:
    74→// imports the grammar (types.ts), the frame truth (frame.ts) and two
    75→// whiteboard siblings (svg) — never the reverse.

   129→/** HTML-escape for text content AND attribute values — the hard gate that
   130→ *  keeps author strings (payload text, icon path data) inside their
   131→ *  context. Applied AFTER sanitizeSvg on icon paths (sanitize first —
   132→ *  strips the payload markup; escape second — nothing breaks out). */
   133→function esc(s: string): string {
   134→  return s
   135→    .replace(/&/g, "&amp;")
   136→    .replace(/</g, "&lt;")
   137→    .replace(/>/g, "&gt;")
   138→    .replace(/"/g, "&quot;")
   139→    .replace(/'/g, "&#39;");
   140→}
   141→
   142→/** Deterministic number → attribute string (6 decimals, no -0, no
   143→ *  exponent noise). The applier recomputes FROM these baked values, so
   144→ *  what the skeleton says is what the frame math uses. */
   145→function fmt(n: number): string {
   146→  return String(Math.round(n * 1e6) / 1e6);
   147→}

   149→/** The counter's rendered string — exported so the pins and any future
   150→ *  surface share ONE formatter. decimals defaults to 0, unit rides after. */
   151→export function formatCounter(value: number, decimals?: number, unit?: string): string {
   152→  return value.toFixed(decimals ?? 0) + (unit ?? "");
   153→}

   159→type RevealMechanism = "bar" | "dash" | "grow" | "wipe";
   160→
   161→const SHAPE_PROGRESS_MECHANISM: Partial<Record<string, "wipe" | "grow" | "dash">> = {
   162→  wipe: "wipe",
   163→  grow: "grow",
   164→  draw: "dash",
   165→};
   166→
   167→/** Shapes pick the mechanism from the FIRST progress-driving motion in
   168→ *  slot order (enter → exit → emphasis); bars/icons carry their native
   169→ *  mechanism always; everything else none. */
   170→function revealMechanism(el: GraphicsElement): RevealMechanism | null {
   171→  if (el.kind === "bar" || el.kind === "barChart") return "bar";
   172→  if (el.kind === "icon") return "dash";
   173→  if (el.kind === "shape") {
   174→    for (const m of [el.enter, el.exit, el.emphasis]) {
   175→      if (m === undefined) continue;
   176→      const mech = SHAPE_PROGRESS_MECHANISM[m.kind];
   177→      if (mech !== undefined) return mech;
   178→    }
   179→  }
   180→  return null;
   181→}

   189→interface TextOpts {
   190→  lines: string[];
   191→  sizePx: number;
   192→  weight: number;
   193→  color: string;
   194→  align: "left" | "center" | "right";
   195→  extraAttrs?: string;
   196→  /** Pre-formatted single-line content (the counter's poster value). */
   197→  content?: string;
   198→}
   199→
   200→/** The shared text block: tspans per line, block centered on the anchor
   201→ *  (dominant-baseline middle), align → text-anchor. */
   202→function textBlock(o: TextOpts): string {
   203→  const anchor = o.align === "left" ? "start" : o.align === "right" ? "end" : "middle";
   204→  const lh = o.sizePx * GRAPHICS_COMPOSE_DEFAULTS.textLineHeight;
   205→  const n = o.lines.length;
   206→  const tspans = o.content !== undefined && n === 1
   207→    ? esc(o.content)
   208→    : o.lines
   209→        .map((line, i) => `<tspan x="0" y="${fmt((i - (n - 1) / 2) * lh)}">${esc(line)}</tspan>`)
   210→        .join("");
   211→  return (
   212→    `<text text-anchor="${anchor}" dominant-baseline="middle"` +
   213→    ` font-size="${fmt(o.sizePx)}" font-weight="${o.weight}" fill="${o.color}"` +
   214→    ` xml:space="preserve"${o.extraAttrs ?? ""}>${tspans}</text>`
   215→  );
   216→}
   217→
   218→function textInner(el: GraphicsElement): string {
   219→  const p = el.text;
   220→  if (p === undefined) return ""; // absent payload → empty element (validator warned)
   221→  return textBlock({
   222→    lines: p.text.split("\n"),
   223→    sizePx: p.sizePx ?? GRAPHICS_COMPOSE_DEFAULTS.sizePx,
   224→    weight: p.weight ?? GRAPHICS_COMPOSE_DEFAULTS.weight,
   225→    color: p.color ?? GRAPHICS_COMPOSE_DEFAULTS.ink,
   226→    align: p.align ?? "center",
   227→  });
   228→}

   230→function counterInner(el: GraphicsElement): string {
   231→  const p = el.counter;
   232→  if (p === undefined) return "";
   233→  const d = p.decimals ?? 0;
   234→  return textBlock({
   235→    lines: [""],
   236→    sizePx: GRAPHICS_COMPOSE_DEFAULTS.sizePx,
   237→    weight: GRAPHICS_COMPOSE_DEFAULTS.weight,
   238→    color: GRAPHICS_COMPOSE_DEFAULTS.ink,
   239→    align: "center",
   240→    extraAttrs: ` data-counter="1" data-decimals="${d}" data-unit="${esc(p.unit ?? "")}"`,
   241→    content: formatCounter(p.to, p.decimals, p.unit), // poster = the final value
   242→  });
   243→}

   245→function barInner(el: GraphicsElement, doc: GraphicsDoc): string {
   246→  const p = el.bar;
   247→  if (p === undefined) return "";
   248→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   249→  const boxW = doc.width * D.chartBoxFraction;
   250→  const boxH = doc.height * D.chartBoxFraction;
   251→  const left = el.at.x - boxW / 2;
   252→  const top = el.at.y - boxH / 2;
   253→  // scaleMax: the payload ceiling, else the data max (all-zero data → 1,
   254→  // so zero-length bars stay honest instead of dividing by zero).
   255→  const dataMax = Math.max(...p.data.map((r) => r.value), 0);
   256→  const scaleMax = p.max !== undefined && p.max > 0 ? p.max : dataMax > 0 ? dataMax : 1;
   257→  const ink = GRAPHICS_COMPOSE_DEFAULTS.ink;
   258→  const n = p.data.length;
   259→  const parts: string[] = [];
   260→  if (p.horizontal === true) {
   261→    const slotH = boxH / n;
   262→    const barH = slotH * D.barSlotFill;
   263→    for (let i = 0; i < n; i++) {
   264→      const row = p.data[i];
   265→      if (row === undefined) continue;
   266→      const len = (row.value / scaleMax) * boxW;
   267→      const cy = top + i * slotH + slotH / 2;
   268→      parts.push(
   269→        `<rect x="${fmt(left)}" y="${fmt(cy - barH / 2)}" width="${fmt(len)}" height="${fmt(barH)}"` +
   270→          ` fill="${ink}" data-len="${fmt(len)}" data-axis="h"/>`,
   271→      );
   272→      parts.push(
   273→        `<text x="${fmt(left - 12)}" y="${fmt(cy)}" text-anchor="end" dominant-baseline="middle"` +
   274→          ` font-size="${D.barLabelSizePx}" fill="${ink}" xml:space="preserve">${esc(row.label + (p.unit ? " " + p.unit : ""))}</text>`,
   275→      );
   276→    }
   277→  } else {
   278→    const slotW = boxW / n;
   279→    const barW = slotW * D.barSlotFill;
   280→    const baseline = top + boxH;
   281→    for (let i = 0; i < n; i++) {
   282→      const row = p.data[i];
   283→      if (row === undefined) continue;
   284→      const len = (row.value / scaleMax) * boxH;
   285→      const x = left + i * slotW + (slotW - barW) / 2;
   286→      parts.push(
   287→        `<rect x="${fmt(x)}" y="${fmt(baseline - len)}" width="${fmt(barW)}" height="${fmt(len)}"` +
   288→          ` fill="${ink}" data-len="${fmt(len)}" data-axis="v" data-base="${fmt(baseline)}"/>`,
   289→      );
   290→      parts.push(
   291→        `<text x="${fmt(left + i * slotW + slotW / 2)}" y="${fmt(baseline + D.barLabelGapPx)}"` +
   292→          ` text-anchor="middle" font-size="${D.barLabelSizePx}" fill="${ink}" xml:space="preserve">` +
   293→          `${esc(row.label + (p.unit ? " " + p.unit : ""))}</text>`,
   294→      );
   295→    }
   296→  }
   297→  return parts.join("\n");
   298→}

   300→function iconInner(el: GraphicsElement): string {
   301→  const p = el.icon;
   302→  if (p === undefined) return "";
   303→  const color = p.color ?? GRAPHICS_COMPOSE_DEFAULTS.ink;
   304→  const sw = p.strokeWidth ?? GRAPHICS_COMPOSE_DEFAULTS.iconStrokeWidth;
   305→  // sanitize FIRST (svg.ts strips payload markup from the d string), then
   306→  // escape (nothing breaks the attribute context). The data-len rides the
   307→  // SANITIZED path — what renders is what measures.
   308→  return p.paths
   309→    .map((d) => {
   310→      const clean = sanitizeSvg(d);
   311→      const len = totalLength(pathSegments(clean));
   312→      return (
   313→        `<path d="${esc(clean)}" fill="none" stroke="${color}" stroke-width="${fmt(sw)}"` +
   314→        ` stroke-linecap="round" stroke-linejoin="round"` +
   315→        ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"/>`
   316→      );
   317→    })
   318→    .join("\n");
   319→}

   321→function calloutInner(el: GraphicsElement): string {
   322→  const p = el.callout;
   323→  if (p === undefined) return "";
   324→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   325→  const sizePx = D.calloutSizePx;
   326→  const ink = D.ink;
   327→  const lines = p.text.split("\n");
   328→  const maxChars = Math.max(...lines.map((l) => l.length), 1);
   329→  const w = Math.max(maxChars * sizePx * D.bubbleCharW + D.bubblePadPx * 2, D.bubbleMinW);
   330→  const h = lines.length * sizePx * D.bubbleLineHeight + D.bubblePadPx * 2;
   331→  const parts: string[] = [];
   332→  if (p.pointsAt !== undefined) {
   333→    // Tail from the bubble's bottom edge to the target — LOCAL coords (the
   334→    // group carries `at`; exact when rotate 0 / scale 1, the authored case).
   335→    const dx = p.pointsAt.x - el.at.x;
   336→    const dy = p.pointsAt.y - el.at.y;
   337→    parts.push(`<line x1="0" y1="${fmt(h / 2)}" x2="${fmt(dx)}" y2="${fmt(dy)}" stroke="${ink}" stroke-width="2"/>`);
   338→    parts.push(`<circle cx="${fmt(dx)}" cy="${fmt(dy)}" r="${D.tailDotR}" fill="${ink}"/>`);
   339→  }
   340→  parts.push(
   341→    `<rect x="${fmt(-w / 2)}" y="${fmt(-h / 2)}" width="${fmt(w)}" height="${fmt(h)}" rx="12"` +
   342→      ` fill="#ffffff" stroke="${ink}" stroke-width="2"/>`,
   343→  );
   344→  parts.push(
   345→    textBlock({
   346→      lines,
   347→      sizePx,
   348→      weight: D.weight,
   349→      color: ink,
   350→      align: "center",
   351→    }),
   352→  );
   353→  return parts.join("\n");
   354→}

   356→function shapeInner(el: GraphicsElement): string {
   357→  const p = el.shape;
   358→  if (p === undefined) return "";
   359→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   360→  const stroke = p.stroke ?? D.ink;
   361→  const fill = p.fill ?? "none";
   362→  const sw = p.strokeWidth ?? D.shapeStrokeWidth;
   363→  const mech = revealMechanism(el);
   364→  const base = ` stroke="${stroke}" stroke-width="${fmt(sw)}" data-shape="${p.kind}"`;
   365→  let attrs = "";
   366→  if (mech === "dash") {
   367→    attrs += ` data-fill="1"`; // fill-after-outline: the applier flips it at progress 1
   368→  }
   369→  if (mech === "grow") {
   370→    attrs += ` data-grow="1"`; // the applier recomputes the geometry from the baked dims
   371→  }
   372→  // Full dims ALWAYS baked (data-w/h/r) — the grow branch recomputes from
   373→  // them; the poster attributes carry the placed geometry.
   374→  switch (p.kind) {
   375→    case "rect": {
   376→      const w = p.w ?? D.shapeW;
   377→      const h = p.h ?? D.shapeH;
   378→      attrs += ` data-w="${fmt(w)}" data-h="${fmt(h)}"`;
   379→      if (mech === "dash") {
   380→        const len = 2 * (w + h);
   381→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   382→      }
   383→      if (mech === "wipe") {
   384→        attrs += ` clip-path="url(#ge-${el.id}--wipe)"`;
   385→      }
   386→      return `<rect x="${fmt(-w / 2)}" y="${fmt(-h / 2)}" width="${fmt(w)}" height="${fmt(h)}"` +
   387→        ` fill="${fill}"${base}${attrs}/>`;
   388→    }
   389→    case "line": {
   390→      const w = p.w ?? D.lineW;
   391→      const h = p.h ?? 0;
   392→      attrs += ` data-w="${fmt(w)}" data-h="${fmt(h)}"`;
   393→      if (mech === "dash") {
   394→        const len = Math.hypot(w, h);
   395→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   396→      }
   397→      return `<line x1="${fmt(-w / 2)}" y1="${fmt(-h / 2)}" x2="${fmt(w / 2)}" y2="${fmt(h / 2)}"` +
   398→        ` fill="none"${base}${attrs}/>`;
   399→    }
   400→    case "circle": {
   401→      const r = p.r ?? D.shapeR;
   402→      attrs += ` data-r="${fmt(r)}"`;
   403→      if (mech === "dash") {
   404→        const len = 2 * Math.PI * r;
   405→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   406→      }
   407→      if (mech === "wipe") {
   408→        attrs += ` clip-path="url(#ge-${el.id}--wipe)"`;
   409→      }
   410→      return `<circle cx="0" cy="0" r="${fmt(r)}" fill="${fill}"${base}${attrs}/>`;
   411→    }
   412→  }
   413→}

   415→/** The wipe mechanism's clip — baked INSIDE the group so the applier's
   416→ *  :scope query finds it and the element's user space owns the coords. */
   417→function wipeClip(el: GraphicsElement): string {
   418→  const p = el.shape;
   419→  if (p === undefined) return "";
   420→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   421→  let x: number;
   422→  let y: number;
   423→  let w: number;
   424→  let h: number;
   425→  switch (p.kind) {
   426→    case "rect": {
   427→      w = p.w ?? D.shapeW;
   428→      h = p.h ?? D.shapeH;
   429→      x = -w / 2;
   430→      y = -h / 2;
   431→      break;
   432→    }
   433→    case "line": {
   434→      w = p.w ?? D.lineW;
   435→      h = (p.h ?? 0) + (p.strokeWidth ?? D.shapeStrokeWidth) + 8;
   436→      x = -w / 2;
   437→      y = -h / 2;
   438→      break;
   439→    }
   440→    case "circle": {
   441→      const r = p.r ?? D.shapeR;
   442→      x = -r;
   443→      y = -r;
   444→      w = 2 * r;
   445→      h = 2 * r;
   446→      break;
   447→    }
   448→  }
   449→  return (
   450→    `<clipPath id="ge-${el.id}--wipe"><rect x="${fmt(x)}" y="${fmt(y)}"` +
   451→    ` width="${fmt(w)}" height="${fmt(h)}" data-w="${fmt(w)}"/></clipPath>`
   452→  );
   453→}

   455→function elementInner(el: GraphicsElement, doc: GraphicsDoc, ctx?: ComposeCtx): string {
   456→  switch (el.kind) {
   457→    case "headline":
   458→    case "subhead":
   459→    case "bullet":
   460→      return textInner(el);
   461→    case "counter":
   462→      return counterInner(el);
   463→    case "bar":
   464→    case "barChart":
   465→      return barInner(el, doc);
   466→    case "icon":
   467→      return iconInner(el);
   468→    case "callout":
   469→      return calloutInner(el);
   470→    case "shape":
   471→      return shapeInner(el);
   472→    case "group":
   473→      return ""; // unreachable — elementNode short-circuits groups; children ride the assembly pass
   474→    case "sprite":
   475→      return spriteInner(el, ctx);
   476→  }
   477→}

   484→ *  one entry per frame: sheet = [strip], sequence = [f0..fN], built by the
   485→ *  caller through the import store — the whiteboard loadVideoLibrary
   486→ *  one-seam law) + the missing keys recorded during the pass (the renderer
   487→ *  refuses fail-closed on a non-empty set). */
   488→export interface ComposeCtx {
   489→  spriteAssets?: Map<string, string[]>;
   490→  missingAssets?: Set<string>;
   491→}
   492→
   493→/** The sprite's inner markup — layout "sheet": one nested <svg viewBox>
   494→ *  window over the strip (the applier moves the window); layout
   495→ *  "sequence": stacked <image> nodes, frame 0 visible (the applier
   496→ *  display-toggles — the mouth-swap discipline). Missing/short frame lists
   497→ *  emit NOTHING and record the key (the renderer refuses; the editor
   498→ *  surfaces). */
   499→function spriteInner(el: GraphicsElement, ctx?: ComposeCtx): string {
   500→  const p = el.sprite;
   501→  if (p === undefined) return "";
   502→  const uris = ctx?.spriteAssets?.get(p.assetKey);
   503→  const needed = p.layout === "sheet" ? 1 : p.frames;
   504→  if (!uris || uris.length < needed) {
   505→    ctx?.missingAssets?.add(p.assetKey);
   506→    return "";
   507→  }
   508→  const halfW = p.frameW / 2;
   509→  const halfH = p.frameH / 2;
   510→  if (p.layout === "sheet") {
   511→    // The horizontal-strip law (v1 — grids are a design-doc slice): the
   512→    // sheet's natural width is frames × frameW; the nested svg's viewBox
   513→    // is the WINDOW the applier slides.
   514→    return (
   515→      `<svg x="${fmt(-halfW)}" y="${fmt(-halfH)}" width="${fmt(p.frameW)}" height="${fmt(p.frameH)}"` +
   516→      ` viewBox="0 0 ${fmt(p.frameW)} ${fmt(p.frameH)}" data-sprite="sheet" data-fw="${fmt(p.frameW)}" data-fh="${fmt(p.frameH)}"` +
   517→      ` preserveAspectRatio="xMinYMin slice"><image href="${uris[0]}" width="${fmt(p.frameW * p.frames)}" height="${fmt(p.frameH)}"/></svg>`
   518→    );
   519→  }
   520→  const images: string[] = [];
   521→  for (let i = 0; i < p.frames; i++) {
   522→    images.push(
   523→      `<image data-frame="${i}" href="${uris[i]}" x="${fmt(-halfW)}" y="${fmt(-halfH)}" width="${fmt(p.frameW)}" height="${fmt(p.frameH)}"` +
   524→        (i === 0 ? "" : ` display="none"`) + `/>`
   525→    );
   526→  }
   527→  return images.join("\n");
   528→}
   529→
   530→/** The element wrapper: the frame engine (applier) owns transform +
   531→ *  opacity; the poster carries the placed state. Groups receive their
   532→ *  children's nodes as the body (the assembly pass below builds them). */
   533→function elementNode(el: GraphicsElement, doc: GraphicsDoc, childrenBody = "", ctx?: ComposeCtx): string {
   534→  const transform =
   535→    `translate(${fmt(el.at.x)} ${fmt(el.at.y)}) rotate(${fmt(el.at.rotateDeg)}) scale(${fmt(el.at.scale)})`;
   536→  const opacity = fmt(el.opacity ?? 1);
   537→  const clip = el.kind === "shape" && revealMechanism(el) === "wipe" ? wipeClip(el) : "";
   538→  const inner = el.kind === "group" ? childrenBody : elementInner(el, doc, ctx);
   539→  const body = clip !== "" ? `${clip}\n${inner}` : inner;
   540→  return `<g id="ge-${el.id}" transform="${transform}" opacity="${opacity}">${body}</g>`;
   541→}

   543→// ---------------------------------------------------------------------------
   544→// The placed state + the group claims (S124 — the element editor's truths)
   545→// ---------------------------------------------------------------------------
   546→
   547→/** The group-claim map for one scene — child id → the group id that renders
   548→ *  it. ONE claim truth: the composer's emission pass AND the element
   549→ *  editor's z-order/hit-test laws read THIS map (S124; the inline loop it
   550→ *  replaces was byte-identical — the G1.3 suite pins the skeleton bytes). */
   551→export function graphicsGroupClaims(scene: GraphicsScene): Map<string, string> {
   552→  const claims = new Map<string, string>();
   553→  for (const el of scene.elements) {
   554→    if (el.kind === "group" && el.group !== undefined) {
   555→      for (const child of el.group.children) claims.set(child, el.id);
   556→    }
   557→  }
   558→  return claims;
   559→}

   561→/** The PLACED frame state — the doc's resting truth for one scene: every
   562→ *  element at its placement (at), base opacity, full reveal. This is the
   563→ *  state the skeleton poster bakes (elementNode carries the placed
   564→ *  transform/opacity; the reveal mechanisms bake progress 1), exported so
   565→ *  the element editor plays EXACTLY the poster state through the ONE
   566→ *  applier while editing (S124: edit-mode preview == the poster, pinned).
   567→ *  NOT an animation-time state — frameAt owns motion; this owns rest.
   568→ *  counterValue stays ABSENT: the applier only rewrites counter text when
   569→ *  the field is present, so the poster's baked end value holds. */
   570→export function placedGraphicsFrame(scene: GraphicsScene): GraphicsFrameState {
   571→  return {
   572→    sceneId: scene.id,
   573→    sceneTMs: 0,
   574→    elements: scene.elements.map((el) => ({
   575→      id: el.id,
   576→      opacity: el.opacity ?? 1,
   577→      x: el.at.x,
   578→      y: el.at.y,
   579→      scale: el.at.scale,
   580→      rotateDeg: el.at.rotateDeg,
   581→      progress: 1,
   582→    })),
   583→  };
   584→}

   590→export interface ComposedGraphicsScene {
   591→  html: string;
   592→  /** S126 — the sprite asset keys the compose could NOT resolve (absent
   593→   *  map entry or a short frame list). Absent = an empty array on callers
   594→   *  that pass no context (byte-shape stability for existing consumers). */
   595→  missingAssets: string[];
   596→}
   597→
   598→/** The static skeleton for one scene — the poster frame. PURE: same
   599→ *  (scene, doc, ctx) → byte-identical html (pinned). The renderer feeds it
   600→ *  to page.setContent; the editor to an iframe srcdoc — one compose, two
   601→ *  callers (the wb-page.ts seam law). Sprite bytes ride the ctx (the
   602→ *  import store resolves them — the ONE seam); a compose with no ctx is
   603→ *  byte-identical to the pre-S126 shape for spriteless docs. */
   604→export function composeGraphicsScene(
   605→  scene: GraphicsScene,
   606→  doc: GraphicsDoc,
   607→  opts?: { spriteAssets?: Map<string, string[]> }
   608→): ComposedGraphicsScene {
   609→  const ctx: ComposeCtx = {
   610→    spriteAssets: opts?.spriteAssets,
   611→    missingAssets: new Set<string>(),
   612→  };
   613→  // Group nesting — children ride the group's children-array order inside
   614→  // the group's <g>; an element claimed by two groups lives in the LAST
   615→  // claim (recorded v1 semantics). The claim map is the exported ONE truth
   616→  // (S124): the element editor's laws read the same map.
   617→  const claims = graphicsGroupClaims(scene);
   618→  const emitted = new Set<string>();
   619→  const parts: string[] = [];
   620→  for (const el of scene.elements) {
   621→    if (el.kind === "group") {
   622→      const inner = (el.group?.children ?? [])
   623→        .map((cid) => scene.elements.find((c) => c.id === cid))
   624→        .filter((c): c is GraphicsElement => c !== undefined && !emitted.has(c.id));
   625→      for (const c of inner) emitted.add(c.id);
   626→      const body = inner.map((c) => elementNode(c, doc, "", ctx)).join("\n");
   627→      parts.push(elementNode(el, doc, body, ctx));
   628→      continue;
   629→    }
   630→    if (claims.has(el.id) || emitted.has(el.id)) continue; // lives inside its group
   631→    parts.push(elementNode(el, doc, "", ctx));
   632→  }
   633→  const html =
   634→    `<!doctype html><html><head><meta charset="utf-8"><style>
   635→html,body{margin:0;padding:0;background:${scene.background};overflow:hidden}
   636→svg{display:block}
   637→text{font-family:Arial,Helvetica,sans-serif}
   638→</style></head><body>
   639→<svg id="board" width="${doc.width}" height="${doc.height}" viewBox="0 0 ${doc.width} ${doc.height}" xmlns="http://www.w3.org/2000/svg">
   640→<rect x="0" y="0" width="${doc.width}" height="${doc.height}" fill="${scene.background}"/>
   641→${parts.join("\n")}
   642→</svg></body></html>`;
   643→  return { html, missingAssets: [...(ctx.missingAssets ?? new Set<string>())].sort() };
   644→}
