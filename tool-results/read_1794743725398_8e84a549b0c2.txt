   230→async function test(name: string, fn: () => void | Promise<void>) {
   231→  try {
   232→    await fn();
   233→    passed++;
   234→    console.log(`  ✓ ${name}`);
   235→  } catch (err) {
   236→    failed++;
   237→    const msg = err instanceof Error ? err.message : String(err);
   238→    failures.push(`[${suite}] ${name} — ${msg}`);
   239→    console.log(`  ✗ ${name}\n      ${msg.split("\n")[0]}`);
   240→  }
   241→}

   243→function ok(cond: unknown, msg: string): asserts cond {
   244→  if (!cond) throw new Error(msg);
   245→}
   246→
   247→function eq<T>(actual: T, expected: T, label = "value") {
   248→  const a = JSON.stringify(actual);
   249→  const b = JSON.stringify(expected);
   250→  if (a !== b) throw new Error(`${label}: expected ${b}, got ${a}`);
   251→}
   252→
   253→function approx(actual: number, expected: number, tol: number, label = "number") {
   254→  if (!(Math.abs(actual - expected) <= tol))
   255→    throw new Error(`${label}: expected ${expected} ±${tol}, got ${actual}`);
   256→}
   257→
   258→function oneOf<T>(actual: T, options: readonly T[], label = "value") {
   259→  if (!options.includes(actual))
   260→    throw new Error(`${label}: expected one of ${JSON.stringify(options)}, got ${JSON.stringify(actual)}`);
   261→}
   262→
   263→const statusOf = (report: { checks: { id: string; status: string }[] }, id: string) => {
   264→  const c = report.checks.find((x) => x.id === id);
   265→  if (!c) throw new Error(`check "${id}" missing from report`);
   266→  return c.status;
   267→};

   331→/** Legacy-compat invariant: pass flag always derives from status, and any
   332→ * unmeasured check drags the whole report out of "passed". */
   333→function assertReportInvariants(report: {
   334→  passed: boolean;
   335→  checks: { id: string; status: string; pass: boolean }[];
   336→}) {
   337→  ok(report.checks.length > 0, "report must contain checks");
   338→  for (const c of report.checks)
   339→    eq(c.pass, c.status === "pass", `check ${c.id}: legacy pass flag`);
   340→  const hasUnmeasured = report.checks.some((c) => c.status === "unmeasured");
   341→  if (hasUnmeasured) ok(!report.passed, "any unmeasured check must fail the report");
   342→  else eq(report.passed, report.checks.every((c) => c.status === "pass"), "overall passed");
   343→}
