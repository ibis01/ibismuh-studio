     1→     1→// G1.3 — the graphics composer: static skeleton + ONE applier (S120 §5 /
     2→     2→// Part B). The wb-page.ts seam law applied to the graphics grammar:
     3→     3→//
     4→     4→//   the renderer rasterizes frameAt() through THESE artifacts; the editor
     5→     5→//   plays back frameAt() through THE SAME artifacts. One compose, one
     6→     6→//   applier, two callers — there is no second implementation to drift.
     7→     7→//
     8→     8→// The three artifacts, mirroring whiteboard/wb-page.ts exactly:
     9→     9→//   composeGraphicsScene(scene, doc) → the static HTML/SVG skeleton: the
    10→    10→//     background, one <g id="ge-{id}"> per element in doc order, every
    11→    11→//     piece of per-frame-reactive geometry baked into data-* attributes,
    12→    12→//     static attributes carrying the PLACED state (the poster frame — the
    13→    13→//     applier corrects before any capture).
    14→    14→//   graphicsFramePayload(frame) → the frame truth, serialized for the
    15→    15→//     page (frameAt's GraphicsFrameState, verbatim per element).
    16→    16→//   APPLY_GRAPHICS_FRAME_FN → the ONE applier, a set-only DOM function
    17→    17→//     evaluated by renderer (page.evaluate) and editor (iframe eval)
    18→    18→//     alike. NO CSS animation anywhere — every frame applied by JS
    19→    19→//     (the determinism law, wb-page.ts verbatim).
    20→    20→//
    21→    21→// THE REVEAL CHANNEL — one mechanism per element (the G1.2 contract:
    22→    22→// "wipe/grow/draw expose the reveal-progress channel the composer
    23→    23→// interprets per payload"):
    24→    24→//   bar/barChart → the payload's native mechanism, bar LENGTHS, always
    25→    25→//     baked (progress 1 = full bars; any progress-driving motion —
    26→    26→//     wipe enter, grow exit, emphasis re-trace — modulates them).
    27→    27→//   icon → stroke-dash along the path arc length (svg.ts#pathSegments →
    28→    28→//     totalLength — the S64 geometry, reused, never a second parser),
    29→    29→//     always baked.
    30→    30→//   shape → the FIRST progress-driving motion in slot order (enter →
    31→    31→//     exit → emphasis) picks the mechanism: wipe → a baked clip rect
    32→    32→//     reveals left→right; grow → the shape geometry scales from zero;
    33→    33→//     draw → the outline dashes on (fill appears at progress 1 — the
    34→    34→//     fill-after-outline law). No progress-driving motion → none.
    35→    35→//   text kinds / counter / callout → no mechanism (their legal motion
    36→    36→//     sets drive opacity/offset/scale/count only).
    37→    37→//
    38→    38→// RECORDED v1 DECISIONS (receipts to tune; the names are the contract —
    39→    39→// GRAPHICS_COMPOSE_DEFAULTS is the ONE table):
    40→    40→//   - compose-time defaults: sizePx 48, weight 700, ink #111111, icon
    41→    41→//     strokeWidth 4, shape strokeWidth 4, callout size 28, align center,
    42→    42→//     chart box = ½ surface centered on `at` (types.ts' own comments —
    43→    43→//     the validator clamps, the composer defaults).
    44→    44→//   - bar color is the ink (a per-bar palette is a design-doc slice);
    45→    45→//     bar labels carry the payload unit after the label.
    46→    46→//   - text renders explicit \n lines via tspans; auto-wrap is a design
    47→    47→//     slice, not a drive-by. dominant-baseline centers the block on the
    48→    48→//     anchor; align maps to text-anchor (left→start, center→middle,
    49→    49→//     right→end).
    50→    50→//   - an ABSENT payload composes as an empty element (the validator
    51→    51→//     warned; parse never refuses — the Part F law).
    52→    52→//   - group children nest INSIDE the group's <g> (the group's frame
    53→    53→//     transform/opacity composes with each child's — SVG multiplies
    54→    54→//     subtree opacity for free). Children ride the group's children-array
    55→    55→//     order; an element claimed by two groups lives in the LAST claim
    56→    56→//     (validator dedupes within one group only — recorded semantics).
    57→    57→//   - no camera: graphics is a peer lane; the W5 camera layer stays the
    58→    58→//     whiteboard's. The doc clock IS the timeline.
    59→    59→//   - the applier is IDEMPOTENT BY CONSTRUCTION: every write is an
    60→    60→//     absolute set on a static node — it creates no elements, reads no
    61→    61→//     layout (the W1 getBBox lesson), touches no wall clock, draws no
    62→    62→//     randomness. Full DOM idempotence rides the G1.10 benchmark's
    63→    63→//     frame-hash pin; the unit pins hold the construction laws.
    64→    64→//
    65→    65→// SECURITY (the hostile-icon pin): icon path data is sanitized with
    66→    66→// svg.ts#sanitizeSvg FIRST (payload markup stripped from the d string

    69→    69→// commands on the raw d; the rendered path is the sanitized remainder —
    70→    70→// the budget was an upper bound, sanitize only removes hostile junk.

   129→   129→/** HTML-escape for text content AND attribute values — the hard gate that
   130→   130→ *  keeps author strings (payload text, icon path data) inside their
   131→   131→ *  context. Applied AFTER sanitizeSvg on icon paths (sanitize first —
   132→   132→ *  strips the payload markup; escape second — nothing breaks out). */
   133→   133→function esc(s: string): string {
   134→   134→  return s
   135→   135→    .replace(/&/g, "&amp;")
   136→   136→    .replace(/</g, "&lt;")
   137→   137→    .replace(/>/g, "&gt;")
   138→   138→    .replace(/"/g, "&quot;")
   139→   139→    .replace(/'/g, "&#39;");
   140→   140→}
   141→   141→
   142→   142→/** Deterministic number → attribute string (6 decimals, no -0, no
   143→   143→ *  exponent noise). The applier recomputes FROM these baked values, so
   144→   144→ *  what the skeleton says is what the frame math uses. */
   145→   145→function fmt(n: number): string {
   146→   146→  return String(Math.round(n * 1e6) / 1e6);
   147→   147→}
   148→   148→
   149→   149→/** The counter's rendered string — exported so the pins and any future
   150→   150→ *  surface share ONE formatter. decimals defaults to 0, unit rides after. */
   151→   151→export function formatCounter(value: number, decimals?: number, unit?: string): string {
   152→   152→  return value.toFixed(decimals ?? 0) + (unit ?? "");
   153→   153→}

   155→   155→// ---------------------------------------------------------------------------
   156→   156→// The reveal mechanism — ONE per element (module-private)
   157→   157→// ---------------------------------------------------------------------------
   158→   158→
   159→   159→type RevealMechanism = "bar" | "dash" | "grow" | "wipe";
   160→   160→
   161→   161→const SHAPE_PROGRESS_MECHANISM: Partial<Record<string, "wipe" | "grow" | "dash">> = {
   162→   162→  wipe: "wipe",
   163→   163→  grow: "grow",
   164→   164→  draw: "dash",
   165→   165→};
   166→   166→
   167→   167→/** Shapes pick the mechanism from the FIRST progress-driving motion in
   168→   168→ *  slot order (enter → exit → emphasis); bars/icons carry their native
   169→   169→ *  mechanism always; everything else none. */
   170→   170→function revealMechanism(el: GraphicsElement): RevealMechanism | null {
   171→   171→  if (el.kind === "bar" || el.kind === "barChart") return "bar";
   172→   172→  if (el.kind === "icon") return "dash";
   173→   173→  if (el.kind === "shape") {
   174→   174→    for (const m of [el.enter, el.exit, el.emphasis]) {
   175→   175→      if (m === undefined) continue;
   176→   176→      const mech = SHAPE_PROGRESS_MECHANISM[m.kind];
   177→   177→      if (mech !== undefined) return mech;
   178→   178→    }
   179→   179→  }
   180→   180→  return null;
   181→   181→}

   189→   189→interface TextOpts {
   190→   190→  lines: string[];
   191→   191→  sizePx: number;
   192→   192→  weight: number;
   193→   193→  color: string;
   194→   194→  align: "left" | "center" | "right";
   195→   195→  extraAttrs?: string;
   196→   196→  /** Pre-formatted single-line content (the counter's poster value). */
   197→   197→  content?: string;
   198→   198→}
   199→   199→
   200→   200→/** The shared text block: tspans per line, block centered on the anchor
   201→   201→ *  (dominant-baseline middle), align → text-anchor. */
   202→   202→function textBlock(o: TextOpts): string {
   203→   203→  const anchor = o.align === "left" ? "start" : o.align === "right" ? "end" : "middle";
   204→   204→  const lh = o.sizePx * GRAPHICS_COMPOSE_DEFAULTS.textLineHeight;
   205→   205→  const n = o.lines.length;
   206→   206→  const tspans = o.content !== undefined && n === 1
   207→   207→    ? esc(o.content)
   208→   208→    : o.lines
   209→   209→        .map((line, i) => `<tspan x="0" y="${fmt((i - (n - 1) / 2) * lh)}">${esc(line)}</tspan>`)
   210→   210→        .join("");
   211→   211→  return (
   212→   212→    `<text text-anchor="${anchor}" dominant-baseline="middle"` +
   213→   213→    ` font-size="${fmt(o.sizePx)}" font-weight="${o.weight}" fill="${o.color}"` +
   214→   214→    ` xml:space="preserve"${o.extraAttrs ?? ""}>${tspans}</text>`
   215→   215→  );
   216→   216→}
   217→   217→
   218→   218→function textInner(el: GraphicsElement): string {
   219→   219→  const p = el.text;
   220→   220→  if (p === undefined) return ""; // absent payload → empty element (validator warned)
   221→   221→  return textBlock({
   222→   222→    lines: p.text.split("\n"),
   223→   223→    sizePx: p.sizePx ?? GRAPHICS_COMPOSE_DEFAULTS.sizePx,
   224→   224→    weight: p.weight ?? GRAPHICS_COMPOSE_DEFAULTS.weight,
   225→   225→    color: p.color ?? GRAPHICS_COMPOSE_DEFAULTS.ink,
   226→   226→    align: p.align ?? "center",
   227→   227→  });
   228→   228→}

   232→   232→  if (p === undefined) return "";
   233→   233→  const d = p.decimals ?? 0;
   234→   234→  return textBlock({

   242→   242→  });

   245→   245→function barInner(el: GraphicsElement, doc: GraphicsDoc): string {
   246→   246→  const p = el.bar;
   247→   247→  if (p === undefined) return "";
   248→   248→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   249→   249→  const boxW = doc.width * D.chartBoxFraction;
   250→   250→  const boxH = doc.height * D.chartBoxFraction;
   251→   251→  const left = el.at.x - boxW / 2;
   252→   252→  const top = el.at.y - boxH / 2;
   253→   253→  // scaleMax: the payload ceiling, else the data max (all-zero data → 1,
   254→   254→  // so zero-length bars stay honest instead of dividing by zero).
   255→   255→  const dataMax = Math.max(...p.data.map((r) => r.value), 0);
   256→   256→  const scaleMax = p.max !== undefined && p.max > 0 ? p.max : dataMax > 0 ? dataMax : 1;
   257→   257→  const ink = GRAPHICS_COMPOSE_DEFAULTS.ink;
   258→   258→  const n = p.data.length;
   259→   259→  const parts: string[] = [];
   260→   260→  if (p.horizontal === true) {
   261→   261→    const slotH = boxH / n;
   262→   262→    const barH = slotH * D.barSlotFill;
   263→   263→    for (let i = 0; i < n; i++) {
   264→   264→      const row = p.data[i];
   265→   265→      if (row === undefined) continue;
   266→   266→      const len = (row.value / scaleMax) * boxW;
   267→   267→      const cy = top + i * slotH + slotH / 2;
   268→   268→      parts.push(
   269→   269→        `<rect x="${fmt(left)}" y="${fmt(cy - barH / 2)}" width="${fmt(len)}" height="${fmt(barH)}"` +
   270→   270→          ` fill="${ink}" data-len="${fmt(len)}" data-axis="h"/>`,
   271→   271→      );
   272→   272→      parts.push(
   273→   273→        `<text x="${fmt(left - 12)}" y="${fmt(cy)}" text-anchor="end" dominant-baseline="middle"` +
   274→   274→          ` font-size="${D.barLabelSizePx}" fill="${ink}" xml:space="preserve">${esc(row.label + (p.unit ? " " + p.unit : ""))}</text>`,
   275→   275→      );
   276→   276→    }
   277→   277→  } else {
   278→   278→    const slotW = boxW / n;
   279→   279→    const barW = slotW * D.barSlotFill;
   280→   280→    const baseline = top + boxH;
   281→   281→    for (let i = 0; i < n; i++) {
   282→   282→      const row = p.data[i];
   283→   283→      if (row === undefined) continue;
   284→   284→      const len = (row.value / scaleMax) * boxH;
   285→   285→      const x = left + i * slotW + (slotW - barW) / 2;
   286→   286→      parts.push(
   287→   287→        `<rect x="${fmt(x)}" y="${fmt(baseline - len)}" width="${fmt(barW)}" height="${fmt(len)}"` +
   288→   288→          ` fill="${ink}" data-len="${fmt(len)}" data-axis="v" data-base="${fmt(baseline)}"/>`,
   289→   289→      );
   290→   290→      parts.push(
   291→   291→        `<text x="${fmt(left + i * slotW + slotW / 2)}" y="${fmt(baseline + D.barLabelGapPx)}"` +
   292→   292→          ` text-anchor="middle" font-size="${D.barLabelSizePx}" fill="${ink}" xml:space="preserve">` +
   293→   293→          `${esc(row.label + (p.unit ? " " + p.unit : ""))}</text>`,
   294→   294→      );
   295→   295→    }
   296→   296→  }
   297→   297→  return parts.join("\n");
   298→   298→}

   300→   300→function iconInner(el: GraphicsElement): string {
   301→   301→  const p = el.icon;
   302→   302→  if (p === undefined) return "";
   303→   303→  const color = p.color ?? GRAPHICS_COMPOSE_DEFAULTS.ink;
   304→   304→  const sw = p.strokeWidth ?? GRAPHICS_COMPOSE_DEFAULTS.iconStrokeWidth;
   305→   305→  // sanitize FIRST (svg.ts strips payload markup from the d string), then
   306→   306→  // escape (nothing breaks the attribute context). The data-len rides the
   307→   307→  // SANITIZED path — what renders is what measures.
   308→   308→  return p.paths
   309→   309→    .map((d) => {
   310→   310→      const clean = sanitizeSvg(d);
   311→   311→      const len = totalLength(pathSegments(clean));
   312→   312→      return (
   313→   313→        `<path d="${esc(clean)}" fill="none" stroke="${color}" stroke-width="${fmt(sw)}"` +
   314→   314→        ` stroke-linecap="round" stroke-linejoin="round"` +
   315→   315→        ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"/>`
   316→   316→      );
   317→   317→    })
   318→   318→    .join("\n");
   319→   319→}
   320→   320→
   321→   321→function calloutInner(el: GraphicsElement): string {
   322→   322→  const p = el.callout;
   323→   323→  if (p === undefined) return "";
   324→   324→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   325→   325→  const sizePx = D.calloutSizePx;
   326→   326→  const ink = D.ink;
   327→   327→  const lines = p.text.split("\n");
   328→   328→  const maxChars = Math.max(...lines.map((l) => l.length), 1);
   329→   329→  const w = Math.max(maxChars * sizePx * D.bubbleCharW + D.bubblePadPx * 2, D.bubbleMinW);
   330→   330→  const h = lines.length * sizePx * D.bubbleLineHeight + D.bubblePadPx * 2;
   331→   331→  const parts: string[] = [];
   332→   332→  if (p.pointsAt !== undefined) {
   333→   333→    // Tail from the bubble's bottom edge to the target — LOCAL coords (the
   334→   334→    // group carries `at`; exact when rotate 0 / scale 1, the authored case).
   335→   335→    const dx = p.pointsAt.x - el.at.x;
   336→   336→    const dy = p.pointsAt.y - el.at.y;
   337→   337→    parts.push(`<line x1="0" y1="${fmt(h / 2)}" x2="${fmt(dx)}" y2="${fmt(dy)}" stroke="${ink}" stroke-width="2"/>`);
   338→   338→    parts.push(`<circle cx="${fmt(dx)}" cy="${fmt(dy)}" r="${D.tailDotR}" fill="${ink}"/>`);
   339→   339→  }
   340→   340→  parts.push(
   341→   341→    `<rect x="${fmt(-w / 2)}" y="${fmt(-h / 2)}" width="${fmt(w)}" height="${fmt(h)}" rx="12"` +
   342→   342→      ` fill="#ffffff" stroke="${ink}" stroke-width="2"/>`,
   343→   343→  );
   344→   344→  parts.push(
   345→   345→    textBlock({
   346→   346→      lines,
   347→   347→      sizePx,
   348→   348→      weight: D.weight,
   349→   349→      color: ink,
   350→   350→      align: "center",
   351→   351→    }),
   352→   352→  );
   353→   353→  return parts.join("\n");
   354→   354→}

   356→   356→function shapeInner(el: GraphicsElement): string {
   357→   357→  const p = el.shape;
   358→   358→  if (p === undefined) return "";
   359→   359→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   360→   360→  const stroke = p.stroke ?? D.ink;
   361→   361→  const fill = p.fill ?? "none";
   362→   362→  const sw = p.strokeWidth ?? D.shapeStrokeWidth;
   363→   363→  const mech = revealMechanism(el);
   364→   364→  const base = ` stroke="${stroke}" stroke-width="${fmt(sw)}" data-shape="${p.kind}"`;
   365→   365→  let attrs = "";
   366→   366→  if (mech === "dash") {
   367→   367→    attrs += ` data-fill="1"`; // fill-after-outline: the applier flips it at progress 1
   368→   368→  }
   369→   369→  if (mech === "grow") {
   370→   370→    attrs += ` data-grow="1"`; // the applier recomputes the geometry from the baked dims
   371→   371→  }
   372→   372→  // Full dims ALWAYS baked (data-w/h/r) — the grow branch recomputes from
   373→   373→  // them; the poster attributes carry the placed geometry.
   374→   374→  switch (p.kind) {
   375→   375→    case "rect": {
   376→   376→      const w = p.w ?? D.shapeW;
   377→   377→      const h = p.h ?? D.shapeH;
   378→   378→      attrs += ` data-w="${fmt(w)}" data-h="${fmt(h)}"`;
   379→   379→      if (mech === "dash") {
   380→   380→        const len = 2 * (w + h);
   381→   381→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   382→   382→      }
   383→   383→      if (mech === "wipe") {
   384→   384→        attrs += ` clip-path="url(#ge-${el.id}--wipe)"`;
   385→   385→      }
   386→   386→      return `<rect x="${fmt(-w / 2)}" y="${fmt(-h / 2)}" width="${fmt(w)}" height="${fmt(h)}"` +
   387→   387→        ` fill="${fill}"${base}${attrs}/>`;
   388→   388→    }
   389→   389→    case "line": {
   390→   390→      const w = p.w ?? D.lineW;
   391→   391→      const h = p.h ?? 0;
   392→   392→      attrs += ` data-w="${fmt(w)}" data-h="${fmt(h)}"`;
   393→   393→      if (mech === "dash") {
   394→   394→        const len = Math.hypot(w, h);
   395→   395→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   396→   396→      }
   397→   397→      return `<line x1="${fmt(-w / 2)}" y1="${fmt(-h / 2)}" x2="${fmt(w / 2)}" y2="${fmt(h / 2)}"` +
   398→   398→        ` fill="none"${base}${attrs}/>`;
   399→   399→    }
   400→   400→    case "circle": {
   401→   401→      const r = p.r ?? D.shapeR;
   402→   402→      attrs += ` data-r="${fmt(r)}"`;
   403→   403→      if (mech === "dash") {
   404→   404→        const len = 2 * Math.PI * r;
   405→   405→        attrs += ` stroke-dasharray="${fmt(len)}" data-len="${fmt(len)}"`;
   406→   406→      }
   407→   407→      if (mech === "wipe") {
   408→   408→        attrs += ` clip-path="url(#ge-${el.id}--wipe)"`;
   409→   409→      }
   410→   410→      return `<circle cx="0" cy="0" r="${fmt(r)}" fill="${fill}"${base}${attrs}/>`;
   411→   411→    }
   412→   412→  }
   413→   413→}

   415→   415→/** The wipe mechanism's clip — baked INSIDE the group so the applier's
   416→   416→ *  :scope query finds it and the element's user space owns the coords. */
   417→   417→function wipeClip(el: GraphicsElement): string {
   418→   418→  const p = el.shape;
   419→   419→  if (p === undefined) return "";
   420→   420→  const D = GRAPHICS_COMPOSE_DEFAULTS;
   421→   421→  let x: number;
   422→   422→  let y: number;
   423→   423→  let w: number;
   424→   424→  let h: number;
   425→   425→  switch (p.kind) {
   426→   426→    case "rect": {
   427→   427→      w = p.w ?? D.shapeW;
   428→   428→      h = p.h ?? D.shapeH;
   429→   429→      x = -w / 2;
   430→   430→      y = -h / 2;
   431→   431→      break;
   432→   432→    }
   433→   433→    case "line": {
   434→   434→      w = p.w ?? D.lineW;
   435→   435→      h = (p.h ?? 0) + (p.strokeWidth ?? D.shapeStrokeWidth) + 8;
   436→   436→      x = -w / 2;
   437→   437→      y = -h / 2;
   438→   438→      break;
   439→   439→    }
   440→   440→    case "circle": {
   441→   441→      const r = p.r ?? D.shapeR;
   442→   442→      x = -r;
   443→   443→      y = -r;
   444→   444→      w = 2 * r;
   445→   445→      h = 2 * r;
   446→   446→      break;
   447→   447→    }
   448→   448→  }
   449→   449→  return (
   450→   450→    `<clipPath id="ge-${el.id}--wipe"><rect x="${fmt(x)}" y="${fmt(y)}"` +
   451→   451→    ` width="${fmt(w)}" height="${fmt(h)}" data-w="${fmt(w)}"/></clipPath>`
   452→   452→  );
   453→   453→}

   455→   455→function elementInner(el: GraphicsElement, doc: GraphicsDoc, ctx?: ComposeCtx): string {
   456→   456→  switch (el.kind) {
   457→   457→    case "headline":
   458→   458→    case "subhead":
   459→   459→    case "bullet":
   460→   460→      return textInner(el);
   461→   461→    case "counter":
   462→   462→      return counterInner(el);
   463→   463→    case "bar":
   464→   464→    case "barChart":
   465→   465→      return barInner(el, doc);
   466→   466→    case "icon":
   467→   467→      return iconInner(el);
   468→   468→    case "callout":
   469→   469→      return calloutInner(el);
   470→   470→    case "shape":
   471→   471→      return shapeInner(el);
   472→   472→    case "group":
   473→   473→      return ""; // unreachable — elementNode short-circuits groups; children ride the assembly pass
   474→   474→    case "sprite":
   475→   475→      return spriteInner(el, ctx);
   476→   476→  }
   477→   477→}

   483→   483→/** The compose pass's sprite context: the RESOLVED asset bytes (data URIs —
   484→   484→ *  one entry per frame: sheet = [strip], sequence = [f0..fN], built by the
   485→   485→ *  caller through the import store — the whiteboard loadVideoLibrary
   486→   486→ *  one-seam law) + the missing keys recorded during the pass (the renderer
   487→   487→ *  refuses fail-closed on a non-empty set). */
   488→   488→export interface ComposeCtx {
   489→   489→  spriteAssets?: Map<string, string[]>;
   490→   490→  missingAssets?: Set<string>;
   491→   491→}
   492→   492→
   493→   493→/** The sprite's inner markup — layout "sheet": one nested <svg viewBox>
   494→   494→ *  window over the strip (the applier moves the window); layout
   495→   495→ *  "sequence": stacked <image> nodes, frame 0 visible (the applier
   496→   496→ *  display-toggles — the mouth-swap discipline). Missing/short frame lists
   497→   497→ *  emit NOTHING and record the key (the renderer refuses; the editor
   498→   498→ *  surfaces). */
   499→   499→function spriteInner(el: GraphicsElement, ctx?: ComposeCtx): string {
   500→   500→  const p = el.sprite;
   501→   501→  if (p === undefined) return "";
   502→   502→  const uris = ctx?.spriteAssets?.get(p.assetKey);
   503→   503→  const needed = p.layout === "sheet" ? 1 : p.frames;
   504→   504→  if (!uris || uris.length < needed) {
   505→   505→    ctx?.missingAssets?.add(p.assetKey);
   506→   506→    return "";
   507→   507→  }
   508→   508→  const halfW = p.frameW / 2;
   509→   509→  const halfH = p.frameH / 2;
   510→   510→  if (p.layout === "sheet") {
   511→   511→    // The horizontal-strip law (v1 — grids are a design-doc slice): the
   512→   512→    // sheet's natural width is frames × frameW; the nested svg's viewBox
   513→   513→    // is the WINDOW the applier slides.
   514→   514→    return (
   515→   515→      `<svg x="${fmt(-halfW)}" y="${fmt(-halfH)}" width="${fmt(p.frameW)}" height="${fmt(p.frameH)}"` +
   516→   516→      ` viewBox="0 0 ${fmt(p.frameW)} ${fmt(p.frameH)}" data-sprite="sheet" data-fw="${fmt(p.frameW)}" data-fh="${fmt(p.frameH)}"` +
   517→   517→      ` preserveAspectRatio="xMinYMin slice"><image href="${uris[0]}" width="${fmt(p.frameW * p.frames)}" height="${fmt(p.frameH)}"/></svg>`
   518→   518→    );
   519→   519→  }
   520→   520→  const images: string[] = [];
   521→   521→  for (let i = 0; i < p.frames; i++) {
   522→   522→    images.push(
   523→   523→      `<image data-frame="${i}" href="${uris[i]}" x="${fmt(-halfW)}" y="${fmt(-halfH)}" width="${fmt(p.frameW)}" height="${fmt(p.frameH)}"` +
   524→   524→        (i === 0 ? "" : ` display="none"`) + `/>`
   525→   525→    );
   526→   526→  }
   527→   527→  return images.join("\n");
   528→   528→}

   530→   530→/** The element wrapper: the frame engine (applier) owns transform +
   531→   531→ *  opacity; the poster carries the placed state. Groups receive their
   532→   532→ *  children's nodes as the body (the assembly pass below builds them). */
   533→   533→function elementNode(el: GraphicsElement, doc: GraphicsDoc, childrenBody = "", ctx?: ComposeCtx): string {
   534→   534→  const transform =
   535→   535→    `translate(${fmt(el.at.x)} ${fmt(el.at.y)}) rotate(${fmt(el.at.rotateDeg)}) scale(${fmt(el.at.scale)})`;
   536→   536→  const opacity = fmt(el.opacity ?? 1);
   537→   537→  const clip = el.kind === "shape" && revealMechanism(el) === "wipe" ? wipeClip(el) : "";
   538→   538→  const inner = el.kind === "group" ? childrenBody : elementInner(el, doc, ctx);
   539→   539→  const body = clip !== "" ? `${clip}\n${inner}` : inner;
   540→   540→  return `<g id="ge-${el.id}" transform="${transform}" opacity="${opacity}">${body}</g>`;
   541→   541→}

   547→   547→/** The group-claim map for one scene — child id → the group id that renders

   598→   598→/** The static skeleton for one scene — the poster frame. PURE: same
   599→   599→ *  (scene, doc, ctx) → byte-identical html (pinned). The renderer feeds it
   600→   600→ *  to page.setContent; the editor to an iframe srcdoc — one compose, two
   601→   601→ *  callers (the wb-page.ts seam law). Sprite bytes ride the ctx (the
   602→   602→ *  import store resolves them — the ONE seam); a compose with no ctx is
   603→   603→ *  byte-identical to the pre-S126 shape for spriteless docs. */
   604→   604→export function composeGraphicsScene(
   605→   605→  scene: GraphicsScene,
   606→   606→  doc: GraphicsDoc,
   607→   607→  opts?: { spriteAssets?: Map<string, string[]> }
   608→   608→): ComposedGraphicsScene {
   609→   609→  const ctx: ComposeCtx = {
   610→   610→    spriteAssets: opts?.spriteAssets,
   611→   611→    missingAssets: new Set<string>(),
   612→   612→  };
   613→   613→  // Group nesting — children ride the group's children-array order inside
   614→   614→  // the group's <g>; an element claimed by two groups lives in the LAST
   615→   615→  // claim (recorded v1 semantics). The claim map is the exported ONE truth
   616→   616→  // (S124): the element editor's laws read the same map.
   617→   617→  const claims = graphicsGroupClaims(scene);
   618→   618→  const emitted = new Set<string>();
   619→   619→  const parts: string[] = [];
   620→   620→  for (const el of scene.elements) {
   621→   621→    if (el.kind === "group") {
   622→   622→      const inner = (el.group?.children ?? [])
   623→   623→        .map((cid) => scene.elements.find((c) => c.id === cid))
   624→   624→        .filter((c): c is GraphicsElement => c !== undefined && !emitted.has(c.id));
   625→   625→      for (const c of inner) emitted.add(c.id);
   626→   626→      const body = inner.map((c) => elementNode(c, doc, "", ctx)).join("\n");
   627→   627→      parts.push(elementNode(el, doc, body, ctx));
   628→   628→      continue;
   629→   629→    }
   630→   630→    if (claims.has(el.id) || emitted.has(el.id)) continue; // lives inside its group
   631→   631→    parts.push(elementNode(el, doc, "", ctx));
   632→   632→  }
   633→   633→  const html =
   634→   634→    `<!doctype html><html><head><meta charset="utf-8"><style>
   635→   635→html,body{margin:0;padding:0;background:${scene.background};overflow:hidden}
   636→   636→svg{display:block}
   637→   637→text{font-family:Arial,Helvetica,sans-serif}
   638→   638→</style></head><body>
   639→   639→<svg id="board" width="${doc.width}" height="${doc.height}" viewBox="0 0 ${doc.width} ${doc.height}" xmlns="http://www.w3.org/2000/svg">
   640→   640→<rect x="0" y="0" width="${doc.width}" height="${doc.height}" fill="${scene.background}"/>
   641→   641→${scene.camera !== undefined ? `<g id="gcam">${parts.join("\n")}</g>` : parts.join("\n")}
   642→   642→</svg></body></html>`;
   643→   643→  return { html, missingAssets: [...(ctx.missingAssets ?? new Set<string>())].sort() };
   644→   644→}
