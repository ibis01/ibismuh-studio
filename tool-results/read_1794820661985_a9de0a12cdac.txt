    19→/** W3 (reconciliation §5.3): the role an asset plays in the stroke
    20→ *  grammar — advisory metadata the validator (and later the W4/W5/W7
    21→ *  schedulers) read when present. Optional + tolerant-read: parsers and
    22→ *  docs written before W3 ignore these fields entirely. */
    23→export type AssetRole = "outline" | "fill" | "label" | "arrow" | "emphasis";
    24→
    25→export interface WhiteboardAsset {
    26→  id: string;
    27→  /** Sanitized SVG markup (a fragment usable inside a <g>). */
    28→  svg: string;
    29→  viewBox: { w: number; h: number };
    30→  tags: string[];
    31→  /** W3 additive metadata — all optional, ignored by old parsers. */
    32→  role?: AssetRole;
    33→  /** Pen groups: groupId → member path ids. One group, one pen —
    34→   *  the validator pins stroke-width/linecap consistency inside. */
    35→  penGroups?: Record<string, string[]>;
    36→  /** Explicit path draw order (default: document order). */
    37→  drawOrder?: string[];
    38→  /** Per-asset duration weight for the fill law's W4 successor. */
    39→  durWeight?: number;
    40→  /** Placement framing hint for the W5 camera director. */
    41→  framing?: { pad?: number };
    42→  /** Alternative match names the W2 mention matcher may consult. */
    43→  alts?: string[];
    44→  /** Accessibility description (screen readers, alt text). */
    45→  a11y?: string;
    46→  /** W4 slice 2: "glyph" replaces this asset's <text> elements with the
    47→   *  committed handwriting outlines at registration (the single-face law);
    48→   *  absent/"marker" keeps the marker wipe byte-for-byte. */
    49→  textMode?: "glyph" | "marker";
    50→}

   477→
   478→export interface ExtractedAsset {
   479→  asset: WhiteboardAsset;
   480→  /** Drawable path elements: id + raw path data (document order = draw
   481→   *  order). W3: the pen attributes ride along (stroke / stroke-width /
   482→   *  linecap / fill) — the validator's caps + fill law read them; old
   483→   *  consumers reading {id, d} are untouched (additive, tolerant). */
   484→  paths: { id: string; d: string; stroke?: string; strokeWidth?: number; linecap?: string; fill?: string }[];
   485→  /** Text elements: id + visible content + anchor. */
   486→  texts: { id: string; content: string; x: number; y: number; size: number }[];
   487→  /** S65 (E1): elements excluded from the drawable index, with the budget
   488→   *  law each one broke — surfaced for diagnostics instead of silently
   489→   *  shrinking the drawable set (the E11 upload path rejects outright).
   490→   *  W3: `detail` carries the exact numbers behind the reason. */
   491→  rejected: { kind: "path" | "text"; id: string; reason: "id-charset" | "path-commands"; detail?: string }[];
   492→}
   493→
   494→/** S65 (E1): per-path command budget, counted as EXPANDED commands through
   495→ *  the real grammar (svg.ts countPathCommands — implicit repeats count),
   496→ *  so a polyline of 10,000 implicit points counts as 10,000 — the
   497→ *  flattening cost the timing engine and the rasterizer would both pay
   498→ *  per frame. */
   499→export const WB_PATH_COMMAND_BUDGET = 512;
   500→
   501→/** Extract drawable elements from an asset's markup (trusted, well-formed
   502→ *  library markup — regex extraction; the sanitizer already ran on load).
   503→ *  S65: the id-charset law and the per-path command budget apply HERE —
   504→ *  an element that breaks either never reaches the drawable index, so no
   505→ *  project event can address it and the renderer's namespace seam
   506→ *  (`${assetId}--${pathId}` DOM ids, `assetId::pathId` frame-plan keys)
   507→ *  can never be fed a corrupting id. */
   508→export function extractAsset(asset: WhiteboardAsset): ExtractedAsset {
   509→  const paths: { id: string; d: string }[] = [];
   510→  const texts: ExtractedAsset["texts"] = [];
   511→  const rejected: ExtractedAsset["rejected"] = [];
   512→  const pathRe = /<path\b[^>]*>/g;
   513→  for (const m of asset.svg.matchAll(pathRe)) {
   514→    const tag = m[0];
   515→    // Attribute-anchored regexes (\s prefix): a bare /d="…"/ would match
   516→    // the tail of id="…" and extract the id as path data.
   517→    const id = /\sid="([^"]+)"/.exec(tag)?.[1];
   518→    const d = /\sd="([^"]+)"/.exec(tag)?.[1];
   519→    if (!id || !d) continue;
   520→    if (!validWhiteboardId(id)) {
   521→      rejected.push({ kind: "path", id, reason: "id-charset", detail: `id "${id}" carries characters outside the namespace law` });
   522→      continue;
   523→    }
   524→    const commands = countPathCommands(d);
   525→    if (commands > WB_PATH_COMMAND_BUDGET) {
   526→      rejected.push({ kind: "path", id, reason: "path-commands", detail: `${commands} expanded commands > ${WB_PATH_COMMAND_BUDGET} budget` });
   527→      continue;
   528→    }
   529→    const strokeRaw = /\sstroke="([^"]*)"/.exec(tag)?.[1];
   530→    const swRaw = /\sstroke-width="([-\d.]+)"/.exec(tag)?.[1];
   531→    const capRaw = /\sstroke-linecap="([^"]*)"/.exec(tag)?.[1];
   532→    const fillRaw = /\sfill="([^"]*)"/.exec(tag)?.[1];
   533→    paths.push({
   534→      id,
   535→      d,
   536→      ...(strokeRaw !== undefined ? { stroke: strokeRaw } : {}),
   537→      ...(swRaw !== undefined ? { strokeWidth: Number(swRaw) } : {}),
   538→      ...(capRaw !== undefined ? { linecap: capRaw } : {}),
   539→      ...(fillRaw !== undefined ? { fill: fillRaw } : {}),
   540→    });
   541→  }
   542→  for (const m of asset.svg.matchAll(/<text\b([^>]*)>([^<]*)<\/text>/g)) {
   543→    const attrs = m[1];
   544→    const content = m[2].trim();
   545→    const id = /\sid="([^"]+)"/.exec(attrs)?.[1];
   546→    if (!id || !content) continue;
   547→    if (!validWhiteboardId(id)) {
   548→      rejected.push({ kind: "text", id, reason: "id-charset", detail: `id "${id}" carries characters outside the namespace law` });
   549→      continue;
   550→    }
   551→    texts.push({
   552→      id,
   553→      content,
   554→      x: Number(/\sx="([-\d.]+)"/.exec(attrs)?.[1] ?? 0),
   555→      y: Number(/\sy="([-\d.]+)"/.exec(attrs)?.[1] ?? 0),
   556→      size: Number(/\sfont-size="([-\d.]+)"/.exec(attrs)?.[1] ?? 32),
   557→    });
   558→  }
   559→  return { asset, paths, texts, rejected };
   560→}

   562→/** Load + register the library: sanitize (already at construction), extract,
   563→ *  and feed the pure scheduler's text registries.
   564→ *
   565→ *  S69 (E9 Phase C / E11 §7 gate 7): `uploaded` merges per-video uploaded
   566→ *  assets AFTER the built-ins — the SAME registration path (sanitize on
   567→ *  load, extraction, the scheduler's text registries), so a placement or
   568→ *  event addressing an upload validates and renders through the exact code
   569→ *  the built-ins ride. Id collisions (an upload vs a built-in, or two
   570→ *  uploads) throw — the /asset route prevents them at write time, so a
   571→ *  collision at load is data corruption and says so. */
   572→export function loadLibrary(uploaded: WhiteboardAsset[] = []): {
   573→  assets: Map<string, WhiteboardAsset>;
   574→  extracted: Map<string, ExtractedAsset>;
   575→  libraryIndex: {
   576→    assetIds: Set<string>;
   577→    pathIdsByAsset: Map<string, Set<string>>;
   578→    textIdsByAsset: Map<string, Set<string>>;
   579→  };
   580→  /** W3: uploads the stroke-authored validator EXCLUDED (warn-and-
   581→   *  exclude — the manifest row stays, the asset never registers, and no
   582→   *  document is mutated; this module never sees one). Built-ins never
   583→   *  land here: a built-in failing its own validator throws, because
   584→   *  that is a programming error, not a data condition. */
   585→  warnings: { assetId: string; reasons: string[] }[];
   586→} {
   587→  const assets = new Map<string, WhiteboardAsset>();
   588→  const extracted = new Map<string, ExtractedAsset>();
   589→  const pathIdsByAsset = new Map<string, Set<string>>();
   590→  const textIdsByAsset = new Map<string, Set<string>>();
   591→  const warnings: { assetId: string; reasons: string[] }[] = [];
   592→  const register = (a: WhiteboardAsset, origin: string): void => {
   593→    if (assets.has(a.id)) throw new Error(`duplicate whiteboard asset id: ${a.id} (${origin})`);
   594→    // W4 slice 2 — glyph mode is per-asset metadata: the transform swaps
   595→    // <text> for committed-outline paths BEFORE extraction, so geometry,
   596→    // the scheduler, the page and the camera all ride the ordinary path
   597→    // machinery. No metadata → the asset is byte-identical (the fallback law).
   598→    let effective = a;
   599→    if (a.textMode === "glyph") {
   600→      const glyph = applyGlyphTextMode(a.svg);
   601→      effective = { ...a, svg: glyph.svg };
   602→      // replaced texts leave ex.texts — re-register their anchors so
   603→      // checkReadability and the W5 framing policy still measure them
   604→      for (const t of glyph.replacedAnchors) {
   605→        TEXT_ANCHORS.set(`${a.id}::${t.id}`, { x: t.x, y: t.y, size: t.size });
   606→      }
   607→    }
   608→    // S139 — the sketch pass: every stroke gains a seeded retrace, every
   609→    // filled shape a clipped hatch, BEFORE extraction — the derived paths
   610→    // ride the ordinary schedule/pen/page/camera machinery like authored
   611→    // strokes. Hands are excluded (a hand is not drawn; it draws) — the
   612→    // renderer embeds them verbatim and a dithered hand would look broken.
   613→    if (!(a.id.startsWith("hand-") || a.tags.includes("hand"))) {
   614→      const sk = withSketchPasses(effective.svg, a.id);
   615→      effective = { ...effective, svg: sk.svg };
   616→    }
   617→    const ex = extractAsset(effective);
   618→    // The PAGE seam rides the effective svg too: composeScene (the render
   619→    // engine AND the editor canvas — one source) embeds asset.svg verbatim,
   620→    // so storing the raw markup here would put the glyph contours in the
   621→    // scheduler's work list while the page still carried the <text> — rows
   622→    // for elements that do not exist, and a word that never renders (the
   623→    // W4 glyph proof caught exactly that: pen rows green, pixels empty).
   624→    // Non-glyph assets: effective === a, byte-identical (the fallback law).
   625→    assets.set(a.id, effective);
   626→    extracted.set(a.id, ex);
   627→    pathIdsByAsset.set(a.id, new Set(ex.paths.map((p) => p.id)));
   628→    textIdsByAsset.set(a.id, new Set(ex.texts.map((t) => t.id)));
   629→    // Register the pure scheduler's text metadata.
   630→    TEXT_IDS_BY_ASSET.set(a.id, ex.texts.map((t) => t.id));
   631→    for (const t of ex.texts) {
   632→      TEXT_CHARS_BY_ID.set(`${a.id}::${t.id}`, t.content.length);
   633→      TEXT_ANCHORS.set(`${a.id}::${t.id}`, { x: t.x, y: t.y, size: t.size });
   634→    }
   635→  };
   636→  // W3: every built-in rides its own validator — a failure is a bug in
   637→  // the library itself and says so loudly (the suite pins all-pass).
   638→  for (const a of ASSETS) {
   639→    const report = validateAsset(a);
   640→    if (!report.ok) {
   641→      throw new Error(
   642→        `built-in whiteboard asset "${a.id}" failed the W3 validator: ${report.findings.map((f) => f.reason).join(" · ")}`
   643→      );
   644→    }
   645→    register(a, "built-in library");
   646→  }
   647→  for (const u of uploaded) {
   648→    // The load-time sanitize law applies to uploads too — trusted-or-not,
   649→    // the gate is a property of the pipeline (they were sanitized at upload
   650→    // as well; this is defense in depth, not a second vocabulary).
   651→    const sanitized: WhiteboardAsset = { ...u, svg: sanitizeSvg(u.svg) };
   652→    // W3 — warn-and-exclude (the risk register's over-rejection answer):
   653→    // an upload the validator refuses never reaches the drawable index;
   654→    // the caller learns why through `warnings` and decides (the studio
   655→    // surfaces, the renderer just renders what validated).
   656→    const report = validateAsset(sanitized);
   657→    if (!report.ok) {
   658→      warnings.push({ assetId: u.id, reasons: report.findings.map((f) => f.reason ?? "invalid geometry") });
   659→      continue;
   660→    }
   661→    register(sanitized, "upload manifest");
   662→  }
   663→  return {
   664→    assets,
   665→    extracted,
   666→    libraryIndex: { assetIds: new Set(assets.keys()), pathIdsByAsset, textIdsByAsset },
   667→    warnings,
   668→  };
   669→}
