    99→/** Split narration into sentences: terminate on . ! ? (keeping the
   100→ *  punctuation — and any trailing quotes/brackets — attached), collapse
   101→ *  whitespace, drop empties. Text without terminal punctuation is ONE
   102→ *  sentence. Deterministic. */
   103→export function splitSentences(narration: string): string[] {
   104→  const text = narration.replace(/\s+/g, " ").trim();
   105→  if (!text) return [];
   106→  const parts = text.split(/(?<=[.!?]["')\]]*)\s+/);
   107→  const out = parts.map((s) => s.trim()).filter(Boolean);
   108→  return out.length > 0 ? out : [text];
   109→}
   110→
   111→/** The narration tokenizer — the ANCHOR INDEX SPACE's law: the W4 voice
   112→ *  anchors (voice-anchors.ts) index words in exactly this tokenization, so
   113→ *  the beat windows' per-sentence slices line up with the anchor list.
   114→ *  Exported (not duplicated) — one tokenizer, two consumers. */
   115→export function words(s: string): string[] {
   116→  return s.toLowerCase().split(/[^a-z0-9']+/).filter(Boolean);
   117→}

   119→// ---------------------------------------------------------------------------
   120→// The cue rules — sentence classification + asset mention matching
   121→// ---------------------------------------------------------------------------

   123→/** Sentence-purpose cues, checked in order (emphasis wins over everything:
   124→ *  an exclamation or a stakes phrase owns its sentence). Index 0 defaults
   125→ *  to setup when nothing stronger fires — the opener IS the setup. */
   126→const EMPHASIS_RE = /[!]|\b(key|crucial|critical|most important|remember|the trick|the whole point|never forget)\b/i;
   127→const SETUP_RE = /\b(imagine|picture|let's|lets|consider|suppose|first(ly)?|to start|to begin|welcome|what (if|is|are|do|does|did)|how (do|does|did)|why (do|does|did))\b/i;
   128→const SUBJECT_RE = /\b(is|are|means|refers?|called|known as|defined as|represents|stands for)\b/i;
   129→const ACTION_RE = /\b(then|next|after|now|push(es|ed)?|flows?|moves?|sends?|draws?|writes?|adds?|takes?|turns?|becomes?|starts?|begins?|runs?|grows?|rises?|falls?|beats?|pumps?|launch(es|ed)?)\b/i;
   130→
   131→// S125 (K1) — exported (was module-private): the rig director consumes the
   132→// SAME classifier for its gesture beats ("reusing beat-plan.ts#classifySentence
   133→// instead of duplicating it, one classifier, two consumers" — roadmap §6).
   134→// The cue rules and the precedence below are UNCHANGED; the whiteboard beat
   135→// plan's own outputs are byte-identical.
   136→export function classifySentence(sentence: string, index: number): BeatPurpose {
   137→  if (EMPHASIS_RE.test(sentence)) return "emphasis";
   138→  if (SETUP_RE.test(sentence)) return "setup";
   139→  if (SUBJECT_RE.test(sentence)) return "subject";
   140→  if (ACTION_RE.test(sentence)) return "action";
   141→  if (index === 0) return "setup";
   142→  return "support";
   143→}

   147→// the beat gains its RHETORICAL purpose (what the sentence does in the

   149→// level — all ADDITIVE optional fields, derived deterministically, compiled
   150→// into the existing AnimationEvent grammar (no second timeline grammar).

   155→/** Why the sentence exists in the ARGUMENT — the directive's §2 taxonomy.
   156→ *  A second lens on the same sentence: derived from the same cue family as
   157→ *  the structural purpose plus the comparison/evidence/conclusion cues.
   158→ *  Old plans absent the field (the tolerant-read law); nothing consumes it
   159→ *  for rendering yet — the editor and the LLM planner read it. */
   160→export type RhetoricalPurpose =
   161→  | "setup"
   162→  | "claim"
   163→  | "mechanism"
   164→  | "comparison"
   165→  | "evidence"
   166→  | "conclusion";
   167→
   168→const CONCLUSION_RE = /\b(in short|in summary|the takeaway|bottom line|ultimately|the point is|never forget|remember this|the whole point|that's why|that is why|in the end|the key insight|the lesson|remember the)\b/i;
   169→const COMPARISON_RE = /\b(better|worse|faster|slower|cheaper|unlike|compared|whereas|instead|rather than|versus|vs\b\.?|more than|less than|the difference)\b/i;
   170→const EVIDENCE_RE = /\b(percent|study|studies|research|survey|measured|statistics|twice|half|double|triple|times)\b|%|\breported\b/i;
   171→
   172→/** The rhetorical classifier — PURE, deterministic, same cue-family laws as
   173→ *  classifySentence (checked in order; the last sentence IS the conclusion
   174→ *  slot — the grammar's close card already owns it). */
   175→export function classifyRhetorical(sentence: string, index: number, total: number): RhetoricalPurpose {
   176→  if (total >= 2 && index === total - 1) return "conclusion";
   177→  if (CONCLUSION_RE.test(sentence)) return "conclusion";
   178→  if (COMPARISON_RE.test(sentence)) return "comparison";
   179→  if (EVIDENCE_RE.test(sentence)) return "evidence";
   180→  // The claim outranks the mechanism — the structural classifier's own
   181→  // precedence (subject before action): "The heart IS a pump" is a claim
   182→  // even though it carries the verb-noun "pump".
   183→  if (SUBJECT_RE.test(sentence)) return "claim";
   184→  if (ACTION_RE.test(sentence)) return "mechanism";
   185→  if (SETUP_RE.test(sentence) || index === 0) return "setup";
   186→  return "claim";
   187→}

   192→export type BeatEmphasis = "low" | "normal" | "high";

   204→  | "write"

   211→/** The effects pack's frozen id prefix (S142) — the mark-capable assets. */

   214→function markActionOf(fxAssetId: string): BeatAction | undefined {
   215→  if (fxAssetId === "fx-underline") return "underline";
   216→  if (fxAssetId === "fx-circle-mark") return "circle";
   217→  // fx-crossout does not exist yet — the action stays in the vocabulary for
   218→  // the editor and the future slice; the other effects punctuate (reveal).
   219→  return "reveal";

   222→function beatActionOf(
   223→  assetIds: readonly string[],
   224→  geometry: Map<string, AssetGeometry>,
   225→  markAssetId: string | undefined
   226→): BeatAction {
   227→  if (markAssetId) return markActionOf(markAssetId) ?? "reveal";
   228→  let paths = 0;
   229→  let texts = 0;
   230→  for (const id of assetIds) {
   231→    paths += geometry.get(id)?.paths.size ?? 0;
   232→    texts += TEXT_IDS_BY_ASSET.get(id)?.length ?? 0;
   233→  }
   234→  if (paths === 0 && texts === 0) return "hold";
   235→  if (paths === 0 && texts > 0) return "write";
   236→  return "draw";
   237→}

   252→  ["coin", /\b(money|dollar|cost|price|revenue|budget|profit|paid|expensive|cheap|cash|pay)\b/i],

   268→/** Tag-token stopwords: palette noise that would match every sentence. */
   269→const TAG_STOPWORDS = new Set([
   270→  "and", "the", "for", "with", "this", "that", "from", "into",
   271→  "upload", "svg", "icon", "symbol", "concept", "drawing", "illustration",
   272→  "white", "black", "simple", "flat", "line", "art", "outline", "style",

   275→function tagTokens(tags: readonly string[] | undefined): string[] {
   276→  if (!tags) return [];
   277→  const out: string[] = [];
   278→  for (const t of tags) {
   279→    for (const w of t.toLowerCase().split(/[^a-z0-9]+/)) {
   280→      if (w.length >= 3 && !TAG_STOPWORDS.has(w)) out.push(w);
   281→    }
   282→  }
   283→  return out;
   284→}

   286→/** Score how strongly a sentence introduces an asset: a cue-table hit is
   287→ *  the strongest signal (the planner's own vocabulary), tag overlap is the
   288→ *  generic signal (uploads, the benchmark library), silence scores 0. */
   289→function mentionScore(
   290→  sentenceTokens: string[],
   291→  sentence: string,
   292→  assetId: string,
   293→  tags: readonly string[] | undefined,
   294→  meta?: AssetBeatMeta
   295→): number {
   296→  let score = 0;
   297→  const cue = ASSET_CUES.get(assetId);
   298→  if (cue && cue.test(sentence)) score += 2;
   299→  const tokens = new Set(sentenceTokens);
   300→  for (const t of tagTokens(tags)) if (tokens.has(t)) score += 1;
   301→  // S147 — the S142 recorded follow-up: score the asset's ALTS (the
   302→  // effects' phrase vocabulary) into the candidate ranking, and give the
   303→  // emphasis role its edge — but ONLY on an emphasis-class sentence, so
   304→  // the marks sprinkle instead of wallpapering every board.
   305→  const lower = sentence.toLowerCase();
   306→  for (const alt of meta?.alts ?? []) {
   307→    if (alt && lower.includes(alt.toLowerCase())) score += 2;
   308→  }
   309→  if (meta?.role === "emphasis" && EMPHASIS_RE.test(sentence)) score += 2;
   310→  return score;
   311→}

   371→/** The exact claimed-key grammar the fill law uses (timing.ts:66-70). */
   372→function claimedKeys(scene: WhiteboardScene): Set<string> {
   373→  const claimed = new Set<string>();
   374→  for (const e of scene.animations) {
   375→    if (e.type === "draw_path") claimed.add(`${e.assetId}::${e.pathId}`);
   376→    else if (e.type === "write_text") claimed.add(`${e.assetId}::text:${e.textId}`);
   377→  }
   378→  return claimed;
   379→}

   381→/** Whether a placement still carries drawable work the beat layer may
   382→ *  schedule (unclaimed paths or texts). */
   383→function hasUnclaimedWork(
   384→  assetId: string,
   385→  scene: WhiteboardScene,
   386→  geometry: Map<string, AssetGeometry>,
   387→  claimed: Set<string>
   388→): boolean {
   389→  const geo = geometry.get(assetId);
   390→  if (geo) {
   391→    for (const pathId of geo.paths.keys()) {
   392→      if (!claimed.has(`${assetId}::${pathId}`)) return true;
   393→    }
   394→  }
   395→  const texts = TEXT_IDS_BY_ASSET.get(assetId) ?? [];
   396→  for (const textId of texts) {
   397→    if (!claimed.has(`${assetId}::text:${textId}`)) return true;
   398→  }
   399→  return false;
   400→}

   402→/**
   403→ * Plan the scene's beats. PURE. Returns [] when the plan degenerates —
   404→ * no sentences, no unclaimed drawables, a clock too short to beat — and
   405→ * the caller falls back to the fill law (the fallback-renders law).
   406→ *
   407→ * Assignment: placements are visited in ORDER (the fill law's own sort:
   408→ * the `order` field, ties by array position — stable, deterministic).
   409→ * Each placement joins the best-scoring sentence at or AFTER the previous
   410→ * placement's sentence (the order-monotonic law): title text draws with
   411→ * the opening sentence, the coin draws when the narrator says "money",
   412→ * labels ride the sentence that names them — but never ahead of an
   413→ * earlier-ordered placement. Unmatched placements stay grouped at the
   414→ * cursor.
   415→ */
   416→export function planBeats(
   417→  scene: WhiteboardScene,
   418→  opts: PlanBeatsOptions
   419→): WhiteboardBeat[] {
   420→  const sentences = splitSentences(scene.narration);
   421→  if (sentences.length === 0) return [];
   422→  if (!(opts.durationMs > 0)) return [];
   423→
   424→  const claimed = claimedKeys(scene);
   425→  const placements = [...scene.placements]
   426→    .sort((a, b) => (a.order ?? 0) - (b.order ?? 0))
   427→    .filter((pl) => hasUnclaimedWork(pl.assetId, scene, opts.geometry, claimed));
   428→  if (placements.length === 0) return [];
   429→
   430→  const sentenceTokens = sentences.map((s) => words(s));
   431→
   432→  // Assign every work-carrying placement to a sentence (order-monotonic,
   433→  // best mention wins; ties → the earliest sentence).
   434→  const bySentence = new Map<number, string[]>();
   435→  let cursor = 0;
   436→  for (const pl of placements) {
   437→    let target: number;
   438→    if (CARD_FIRST.has(pl.assetId)) {
   439→      target = 0;
   440→    } else if (CARD_LAST.has(pl.assetId)) {
   441→      target = sentences.length - 1;
   442→    } else {
   443→      target = cursor;
   444→      let best = 0;
   445→      for (let i = cursor; i < sentences.length; i++) {
   446→        const score = mentionScore(
   447→          sentenceTokens[i],
   448→          sentences[i],
   449→          pl.assetId,
   450→          opts.assetTags?.get(pl.assetId),
   451→          opts.assetMeta?.get(pl.assetId)
   452→        );
   453→        if (score > best) {
   454→          best = score;
   455→          target = i;
   456→        }
   457→      }
   458→    }
   459→    target = Math.min(Math.max(target, cursor), sentences.length - 1);
   460→    const bucket = bySentence.get(target) ?? [];
   461→    bucket.push(pl.assetId);
   462→    bySentence.set(target, bucket);
   463→    cursor = Math.max(cursor, target);
   464→  }
   465→  if (bySentence.size === 0) return [];
   466→
   467→  // W4 — the windows. Two sources, one law:
   468→  //   aligned anchors → each sentence's window is its words' spoken span
   469→  //   (the clock the VO actually spoke; the hold extends to the scene
   470→  //   clock's edge);
   471→  //   absent / mismatched / degenerate → the word-proportional estimator
   472→  //   across the fill law's envelope ([LEAD_IN, duration − HOLD]),
   473→  //   byte-for-byte the W2 behavior. Anchors are advisory inputs — a
   474→  //   misheard ASR pass degrades, it never corrupts the choreography.
   475→  const narrationWords = sentences.map((s) => words(s));
   476→  const flatWordCount = narrationWords.reduce((a, w) => a + w.length, 0);
   477→  const env = anchorEnvelope(opts.anchors, opts.durationMs);
   478→  const sortedWords =
   479→    opts.anchors && env ? [...opts.anchors.byWord].sort((a, b) => a.index - b.index) : null;
   480→  let windows: { startMs: number; endMs: number }[] | null = null;
   481→  let anchored = false;
   482→  if (sortedWords && sortedWords.length === flatWordCount) {
   483→    const candidate: { startMs: number; endMs: number }[] = [];
   484→    let off = 0;
   485→    let sane = true;
   486→    for (const toks of narrationWords) {
   487→      const first = sortedWords[off];
   488→      const last = sortedWords[off + toks.length - 1];
   489→      off += toks.length;
   490→      if (!first || !last || last.endMs <= first.startMs) {
   491→        sane = false;
   492→        break;
   493→      }
   494→      candidate.push({ startMs: Math.round(first.startMs), endMs: Math.round(last.endMs) });
   495→    }
   496→    // Windows must march forward (weird ASR spans never reorder the beats).
   497→    for (let i = 1; sane && i < candidate.length; i++) {
   498→      if (candidate[i].startMs < candidate[i - 1].endMs) sane = false;
   499→    }
   500→    if (sane) {
   501→      windows = candidate;
   502→      anchored = true;
   503→    }
   504→  }
   505→  if (!windows) {
   506→    // Word-proportional windows across the draw span (the fill law's
   507→    // envelope: [LEAD_IN, duration − HOLD], floored — timing.ts:102).
   508→    const weights = sentences.map((s) => Math.max(words(s).length, MIN_BEAT_WORDS));
   509→    const totalWeight = weights.reduce((a, b) => a + b, 0);
   510→    const drawSpan = Math.max(MIN_DRAW_SPAN_MS, opts.durationMs - LEAD_IN_MS - HOLD_END_MS);
   511→    let t = LEAD_IN_MS;
   512→    windows = weights.map((w) => {
   513→      const start = t;
   514→      t += (w / totalWeight) * drawSpan;
   515→      return { startMs: Math.round(start), endMs: Math.round(t) };
   516→    });
   517→  }
   518→
   519→  const beats: WhiteboardBeat[] = [];
   520→  for (const [sentence, assetIds] of bySentence) {
   521→    const window = windows[sentence];
   522→    const purpose = classifySentence(sentences[sentence], sentence);
   523→    // S147 — the additive fields. The mark: the FIRST fx-* asset assigned to
   524→    // a HIGH-emphasis beat (deterministic; a mark on a normal beat draws in
   525→    // order — the sprinkle law). The action rides the mark when one fires.
   526→    const emphasis = beatEmphasisOf(purpose);
   527→    const markAssetId =
   528→      emphasis === "high"
   529→        ? assetIds.find((id) => id.startsWith(FX_PREFIX))
   530→        : undefined;
   531→    beats.push({
   532→      id: `beat-${sentence + 1}`,
   533→      purpose,
   534→      sentence,
   535→      phrase: sentences[sentence],
   536→      assetIds,
   537→      startMs: window.startMs,
   538→      endMs: window.endMs,
   539→      rhetorical: classifyRhetorical(sentences[sentence], sentence, sentences.length),
   540→      action: beatActionOf(assetIds, opts.geometry, markAssetId),
   541→      emphasis,
   542→      markAssetId,
   543→    });
   544→  }
   545→  beats.sort((a, b) => a.sentence - b.sentence);
   546→
   547→  // R1 — the opening-stroke guarantee (see the two thresholds): when the
   548→  // earliest asset-carrying beat opens past its world's threshold, its
   549→  // FIRST placement (draw order) pulls forward to sentence 0's window; the
   550→  // rest of its mentions keep their sentence. A closer-card-only scene
   551→  // stays put — pulling a close card to the opening would invert the
   552→  // closer's design. Deterministic.
   553→  const firstBeat = beats[0];
   554→  const closerOnly =
   555→    firstBeat.sentence === sentences.length - 1 &&
   556→    beats.length === 1 &&
   557→    firstBeat.assetIds.every((id) => CARD_LAST.has(id));
   558→  const openingMax = anchored ? OPENING_STROKE_MAX_MS : OPENING_STROKE_ESTIMATOR_MAX_MS;
   559→  if (firstBeat.sentence > 0 && firstBeat.startMs > openingMax && !closerOnly) {
   560→    const moved = firstBeat.assetIds.shift();
   561→    if (firstBeat.assetIds.length === 0) beats.shift();
   562→    if (moved) {
   563→      const purpose = classifySentence(sentences[0], 0);
   564→      const emphasis = beatEmphasisOf(purpose);
   565→      const markAssetId =
   566→        emphasis === "high" && moved.startsWith(FX_PREFIX) ? moved : undefined;
   567→      beats.unshift({
   568→        id: "beat-1",
   569→        purpose,
   570→        sentence: 0,
   571→        phrase: sentences[0],
   572→        assetIds: [moved],
   573→        startMs: windows[0].startMs,
   574→        endMs: windows[0].endMs,
   575→        rhetorical: classifyRhetorical(sentences[0], 0, sentences.length),
   576→        action: beatActionOf([moved], opts.geometry, markAssetId),
   577→        emphasis,
   578→        markAssetId,
   579→      });
   580→    }
   581→  }
   582→  return beats;
   583→}

   585→// ---------------------------------------------------------------------------
   586→// compileBeats — beats → AnimationEvent[] (the doc's existing grammar)
   587→// ---------------------------------------------------------------------------
   588→
   589→/**
   590→ * Compile a beat plan into the scene's event list: the author's explicit
   591→ * events pass through untouched (senior — the precedence law), every beat
   592→ * fills its window with the unclaimed drawables of its placements, using
   593→ * the fill law's own arithmetic (weight = scaled stroke length / text
   594→ * char count; per-item clamps; the stroke gap; a uniform fit factor).
   595→ * An item never outlasts the scene clock. PURE.
   596→ */
   597→export function compileBeats(
   598→  beats: WhiteboardBeat[],
   599→  scene: WhiteboardScene,
   600→  durationMs: number,
   601→  geometry: Map<string, AssetGeometry>
   602→): AnimationEvent[] {
   603→  const events: AnimationEvent[] = [...scene.animations];
   604→  const claimed = claimedKeys(scene);
   605→
   606→  for (const beat of beats) {
   607→    interface Item {
   608→      kind: "path" | "text";
   609→      assetId: string;
   610→      id: string;
   611→      weight: number;
   612→    }
   613→    const items: Item[] = [];
   614→    for (const assetId of beat.assetIds) {
   615→      const geo = geometry.get(assetId);
   616→      if (geo) {
   617→        const scale = scene.placements.find((pl) => pl.assetId === assetId)?.placement.scale ?? 1;
   618→        for (const [pathId, segs] of geo.paths) {
   619→          if (claimed.has(`${assetId}::${pathId}`)) continue;
   620→          items.push({ kind: "path", assetId, id: pathId, weight: Math.max(120, totalLength(segs) * scale) });
   621→        }
   622→      }
   623→      const texts = TEXT_IDS_BY_ASSET.get(assetId) ?? [];
   624→      for (const textId of texts) {
   625→        if (claimed.has(`${assetId}::text:${textId}`)) continue;
   626→        const chars = TEXT_CHARS_BY_ID.get(`${assetId}::${textId}`) ?? 12;
   627→        items.push({ kind: "text", assetId, id: textId, weight: 40 * chars });
   628→      }
   629→    }
   630→    if (items.length === 0) continue; // the every-beat-has-an-action law
   631→
   632→    // S147 — the emphasis mark fires LAST: the labeled item lands, THEN the
   633→    // mark punctuates it (the W4 law — emphasis fires after the spoken
   634→    // claim). Stable sort: only mark-asset items move; absent markAssetId
   635→    // leaves the order byte-identical to the pre-S147 compile.
   636→    if (beat.markAssetId) {
   637→      items.sort(
   638→        (a, b) =>
   639→          (a.assetId === beat.markAssetId ? 1 : 0) - (b.assetId === beat.markAssetId ? 1 : 0)
   640→      );
   641→    }
   642→
   643→    // The fill law's arithmetic, scoped to the beat window.
   644→    const gaps = PATH_GAP_MS * (items.length - 1);
   645→    const avail = Math.max(120, beat.endMs - beat.startMs - gaps);
   646→    const totalWeight = items.reduce((a, w) => a + w.weight, 0);
   647→    let rawSum = 0;
   648→    const raw = items.map((w) => {
   649→      const ideal = (w.weight / totalWeight) * avail;
   650→      const d =
   651→        w.kind === "path"
   652→          ? Math.max(MIN_PATH_MS, Math.min(MAX_PATH_MS, ideal))
   653→          : Math.max(500, (TEXT_CHARS_BY_ID.get(`${w.assetId}::${w.id}`) ?? 12) * MS_PER_CHAR);
   654→      rawSum += d;
   655→      return d;
   656→    });
   657→    const f = rawSum > 0 ? Math.min(2.8, Math.max(0.5, avail / rawSum)) : 1;
   658→    let t = beat.startMs;
   659→    for (let i = 0; i < items.length; i++) {
   660→      const w = items[i];
   661→      // Never outlast the scene clock (the hold is the window's own edge).
   662→      const dur = Math.max(120, Math.min(Math.round(raw[i] * f), Math.max(120, durationMs - t)));
   663→      if (w.kind === "path") {
   664→        events.push({ type: "draw_path", assetId: w.assetId, pathId: w.id, startMs: Math.round(t), durationMs: dur });
   665→        claimed.add(`${w.assetId}::${w.id}`);
   666→      } else {
   667→        events.push({ type: "write_text", assetId: w.assetId, textId: w.id, startMs: Math.round(t), durationMs: dur });
   668→        claimed.add(`${w.assetId}::text:${w.id}`);
   669→      }
   670→      t += dur + PATH_GAP_MS;
   671→    }
   672→  }
   673→  return events.sort((a, b) => a.startMs - b.startMs);
   674→}

   676→// ---------------------------------------------------------------------------
   677→// beatSceneEvents — the seam the render and the editor preview share
   678→// ---------------------------------------------------------------------------
   679→
   680→/**
   681→ * The W2 scheduling entry point, consumed by BOTH the renderer
   682→ * (render.ts) and the editor preview (wb-canvas.tsx) — the one-source
   683→ * law: same inputs, same beats, same frames on both surfaces.
   684→ *
   685→ * Plan → compile; a degenerate plan (no beats, no compiled work beyond
   686→ * the author's events) falls back to `scheduleScene` BYTE-FOR-BYTE — the
   687→ * fallback-renders law. When the plan is usable, explicit events stay
   688→ * senior inside compileBeats and the compiled fills ride the narration's
   689→ * windows.
   690→ */
   691→export function beatSceneEvents(
   692→  scene: WhiteboardScene,
   693→  durationMs: number,
   694→  geometry: Map<string, AssetGeometry>,
   695→  assetTags?: ReadonlyMap<string, readonly string[]>,
   696→  anchors?: WordAnchors,
   697→  assetMeta?: ReadonlyMap<string, AssetBeatMeta>
   698→): AnimationEvent[] {
   699→  const fallback = () => scheduleScene(scene, durationMs, geometry, anchors);
   700→  if (!(durationMs > 0)) return fallback();
   701→  const beats = planBeats(scene, { durationMs, geometry, assetTags, anchors, assetMeta });
   702→  if (beats.length === 0) return fallback();
   703→  const events = compileBeats(beats, scene, durationMs, geometry);
   704→  // A usable plan always adds the unclaimed work; an empty list here is
   705→  // the degenerate corner (nothing drawable at all) — the fill law's
   706→  // output would be identical, so either path agrees. Take the fallback
   707→  // for one-source consistency.
   708→  if (events.length === 0) return fallback();
   709→  return events;
   710→}
